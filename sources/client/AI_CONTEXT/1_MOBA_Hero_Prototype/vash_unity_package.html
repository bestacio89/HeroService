using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string heroId      = "vash";
    public string displayName = "Vash the Stampede";
    public HeroRole role      = HeroRole.Marksman;

    [Header("Core progression — Niveau 1 à 15")]
    public ScaledStat maxHp          = new(2650f,  5800f);
    public ScaledStat hpRegen        = new(7.8f,   16.2f);
    public ScaledStat maxMana        = new(500f,   1900f);
    public ScaledStat manaRegen      = new(5.0f,    8.5f);
    public ScaledStat physicalAttack = new(112f,    235f);
    public ScaledStat magicPower     = new(0f,        0f);
    public ScaledStat physicalDefense= new(16f,      52f);
    public ScaledStat magicDefense   = new(18f,      58f);
    public ScaledStat attackSpeed    = new(1.20f,   1.75f);

    [Header("Fixed hero properties")]
    public float moveSpeed         = 270f;
    public float basicAttackRange  = 9.0f;
    public float attackSpeedRatio  = 1.0f;
    public float criticalDamage    = 2.0f;

    [Header("Passif — Mode Sérieux (scaling par niveau)")]
    public ScaledStat seriousAtkBonus   = new(0.18f, 0.32f);  // multiplicateur ATQ
    public ScaledStat seriousAtkSpeed   = new(0.18f, 0.30f);  // bonus flat AS
    public ScaledStat seriousSlowAmount = new(0.15f, 0.25f);  // slow% tir basic

    [Header("Angel Arm — coût HP fixe")]
    public float angelArmHpCostPercent = 0.10f;   // 10% HP max à l'usage

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new(
        "Tir .45 Long Colt",
        new[] { 0f },
        new[] { 0f },
        new[] { 0f },
        0f);

    // Skill 1 — Quickdraw Barrage : dégâts PAR TIR, 4 tirs normaux / 6 tirs Sérieux
    public AbilityRankData skillOne = new(
        "Quickdraw Barrage",
        new[] { 52f, 68f, 84f, 100f },
        new[] { 55f, 62f, 68f,  75f },
        new[] {  8f,  7f,  6f,   5f },
        0.45f);

    // Skill 2 — Dodge & Volley : dégâts PAR TIR, 3 tirs normaux / 5 tirs Sérieux
    public AbilityRankData skillTwo = new(
        "Dodge and Volley",
        new[] {  70f,  92f, 114f, 136f },
        new[] {  70f,  78f,  86f,  94f },

    public RuntimeHeroStats CreateRuntime(int level)

        return new RuntimeHeroStats
        {
            level            = level,
            maxHp            = maxHp.Evaluate(level),
            hpRegen          = hpRegen.Evaluate(level),

            manaRegen        = manaRegen.Evaluate(level),
            physicalAttack   = physicalAttack.Evaluate(level),
            magicPower       = magicPower.Evaluate(level),

public enum HeroRole
{
    Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin, AssassinFighter
}

[Serializable]
public struct ScaledStat
{
    public float level1;
    public float level15;

    public ScaledStat(float level1, float level15)
    {
        this.level1  = level1;
        this.level15 = level15;
    }

    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        if (level == 1) return level1;
        float t = (level - 1f) / 14f;
        return Mathf.Lerp(level1, level15, t);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] manaCost;
    public float[] cooldown;
    public float   scalingPhysicalAttack;

    public AbilityRankData(string abilityName, float[] baseDamage,
                           float[] manaCost, float[] cooldown, float scalingPhysicalAttack)
    {
        this.abilityName           = abilityName;
        this.baseDamage            = baseDamage;
        this.manaCost              = manaCost;
        this.cooldown              = cooldown;
        this.scalingPhysicalAttack = scalingPhysicalAttack;
    }

    public float DamageAtRank(int rank, float physicalAttack)
        => baseDamage[Mathf.Clamp(rank - 1, 0, baseDamage.Length - 1)]
         + physicalAttack * scalingPhysicalAttack;

    public float ManaAtRank(int rank)
        => manaCost[Mathf.Clamp(rank - 1, 0, manaCost.Length - 1)];

    public float CooldownAtRank(int rank)
        => cooldown[Mathf.Clamp(rank - 1, 0, cooldown.Length - 1)];
}

[Serializable]
public struct RuntimeHeroStats

    public float physicalDefense;
    public float magicDefense;
    public float attackSpeed;
    public float moveSpeed;
    public float basicAttackRange;
    public float attackSpeedRatio;
    public float criticalDamage;
}
C# VashHeroController.cs
// VashHeroController.cs
// Contrôleur principal : mouvement, tir, cooldowns, passif Mode Sérieux, coût HP Angel Arm.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class VashHeroController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator      animator;
    [SerializeField] private NavMeshAgent  agent;
    [SerializeField] private VashCombatVFX combatVfx;
    [SerializeField] private Transform     gunSpawn;
    [SerializeField] private LayerMask     groundMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentMana;
    [SerializeField] private bool  isDead;

    // ── Passif Mode Sérieux ───────────────────────────────────────────
    private int   hitStreak             = 0;
    private float seriousTimer          = 0f;
    private float seriousCooldownTimer  = 0f;
    private bool  isSeriousModeActive   = false;

    // ── Cooldowns ─────────────────────────────────────────────────────
    private float basicCooldown;
    private float skillOneCooldown;
    private float skillTwoCooldown;
    private float ultimateCooldown;

    public RuntimeHeroStats Stats   => stats;
    public float HpPercent          => stats.maxHp   <= 0f ? 0f : currentHp   / stats.maxHp;
    public float ManaPercent        => stats.maxMana <= 0f ? 0f : currentMana / stats.maxMana;
    public bool  IsDead             => isDead;
    public bool  IsSeriousMode      => isSeriousModeActive;

    private void Reset()
    {
        agent    = GetComponent<NavMeshAgent>();
        animator = GetComponentInChildren<Animator>();
    }

    private void Awake()
    {
        if (agent == null) agent = GetComponent<NavMeshAgent>();
        ApplyLevel(heroLevel, true);
    }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        TickSeriousMode();
        RegenerateResources();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats     = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore)
        {
            currentHp   = stats.maxHp;
            currentMana = stats.maxMana;
        }
        else
        {
            currentHp   = Mathf.Min(currentHp,   stats.maxHp);
            currentMana = Mathf.Min(currentMana, stats.maxMana);
        }
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
        animator.SetBool("IsMoving", true);
    }

    // ── Attaque basique — Tir .45 Long Colt ──────────────────────────
    public void BasicAttack(Health target)
    {
        if (target == null || basicCooldown > 0f || isDead) return;

        float bonus  = isSeriousModeActive
                     ? stats.physicalAttack * heroDefinition.seriousAtkBonus.Evaluate(heroLevel)
                     : 0f;
        float damage = stats.physicalAttack + bonus;

        basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
        animator.SetTrigger("BasicAttack");
        combatVfx.PlayBulletTrail(gunSpawn.position, target.transform.position);
        target.ApplyPhysicalDamage(damage);

        hitStreak++;
        if (hitStreak >= 4 && seriousCooldownTimer <= 0f)
            ActivateSeriousMode();
    }

    // ── Skill 1 — Quickdraw Barrage ───────────────────────────────────
    public bool TryCastSkillOne(Transform target)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;

        SpendMana(heroDefinition.skillOne.ManaAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);

        int   shots       = isSeriousModeActive ? 6 : 4;
        float dmgPerShot  = heroDefinition.skillOne.DamageAtRank(skillOneRank, stats.physicalAttack);
        float totalDamage = dmgPerShot * shots;

        agent.isStopped = true;
        transform.LookAt(new Vector3(target.position.x, transform.position.y, target.position.z));
        animator.SetTrigger("CastSkill01");
        combatVfx.PlayQuickdrawBarrage(gunSpawn.position, target.position, shots);
        target.GetComponent<Health>()?.ApplyPhysicalDamage(totalDamage);

        return true;
    }

    // ── Skill 2 — Dodge & Volley ──────────────────────────────────────
    public bool TryCastSkillTwo(Vector3 dashDirection)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;

        SpendMana(heroDefinition.skillTwo.ManaAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);

        StartCoroutine(DodgeVolleyRoutine(dashDirection));
        return true;
    }

    private IEnumerator DodgeVolleyRoutine(Vector3 dashDir)
    {
        agent.isStopped = true;
        combatVfx.SetDodgeInvincible(true);
        animator.SetTrigger("CastSkill02");

        // Dash
        Vector3 dashTarget = transform.position + dashDir.normalized * 4.5f;
        transform.position = dashTarget;
        agent.Warp(dashTarget);
        yield return new WaitForSeconds(0.3f);   // invincibilité
        combatVfx.SetDodgeInvincible(false);

        // Tirs derrière
        int   shots      = isSeriousModeActive ? 5 : 3;
        float dmgPerShot = heroDefinition.skillTwo.DamageAtRank(skillTwoRank, stats.physicalAttack);
        float bonus      = isSeriousModeActive ? 1.3f : 1.0f;
        combatVfx.PlayDodgeVolley(transform.position, -transform.forward, shots);

        // AoE arc arrière
        Collider[] hits = Physics.OverlapSphere(transform.position, 8f);
        foreach (var col in hits)
        {
            float angle = Vector3.Angle(-transform.forward,
                          (col.transform.position - transform.position).normalized);
            if (angle < 50f)
                col.GetComponent<Health>()?.ApplyPhysicalDamage(dmgPerShot * bonus);
        }

        yield return new WaitForSeconds(0.1f);
        agent.isStopped = false;
    }

    // ── Ultime — Angel Arm ────────────────────────────────────────────
    public bool TryCastUltimate(Vector3 direction)
    {
        if (!CanCast(heroDefinition.ultimate, ultimateRank, ultimateCooldown)) return false;

        SpendMana(heroDefinition.ultimate.ManaAtRank(ultimateRank));
        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);

        StartCoroutine(AngelArmRoutine(direction));
        return true;
    }

    private IEnumerator AngelArmRoutine(Vector3 direction)
    {
        agent.isStopped = true;

        // Phase 1 : transformation bras (hors contrôle 0.5s)
        animator.SetTrigger("CastUltimate");
        combatVfx.PlayAngelArmTransform(transform.position);
        yield return new WaitForSeconds(0.5f);

        // Phase 2 : rayon dévastateur
        transform.forward = direction.normalized;
        float mainDmg  = heroDefinition.ultimate.baseDamage[Mathf.Clamp(ultimateRank - 1, 0, 2)];
        float outerDmg = ultimateRank == 1 ? 280f : ultimateRank == 2 ? 420f : 560f;

        combatVfx.PlayAngelArmBeam(transform.position, direction, 22f);

        Collider[] inBeam = Physics.OverlapBox(
            transform.position + direction.normalized * 11f,
            new Vector3(2f, 2f, 11f),
            Quaternion.LookRotation(direction));
        foreach (var col in inBeam)
            col.GetComponent<Health>()?.ApplyPhysicalDamage(mainDmg);

        Collider[] nearBeam = Physics.OverlapSphere(transform.position + direction.normalized * 11f, 6f);
        foreach (var col in nearBeam)
            col.GetComponent<Health>()?.ApplyPhysicalDamage(outerDmg);

        // Coût HP — Vash souffre d'avoir utilisé l'Angel Arm
        float hpCost = stats.maxHp * heroDefinition.angelArmHpCostPercent;
        currentHp    = Mathf.Max(1f, currentHp - hpCost);   // ne peut pas tuer Vash lui-même
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, false);

        yield return new WaitForSeconds(0.6f);
        agent.isStopped = false;
    }

    // ── Passif Mode Sérieux ───────────────────────────────────────────
    private void ActivateSeriousMode()
    {
        isSeriousModeActive   = true;
        seriousTimer          = 6f;
        hitStreak             = 0;
        float cd              = Mathf.Lerp(15f, 11f, (heroLevel - 1f) / 14f);
        seriousCooldownTimer  = cd;
        animator.SetBool("SeriousMode", true);
        combatVfx.SetSeriousMode(true, heroDefinition.seriousAtkBonus.Evaluate(heroLevel));
    }

    private void DeactivateSeriousMode()
    {
        isSeriousModeActive = false;
        animator.SetBool("SeriousMode", false);
        combatVfx.SetSeriousMode(false, 0f);
    }

    private void TickSeriousMode()
    {
        if (seriousCooldownTimer > 0f) seriousCooldownTimer -= Time.deltaTime;
        if (isSeriousModeActive)
        {
            seriousTimer -= Time.deltaTime;
            if (seriousTimer <= 0f) DeactivateSeriousMode();
        }
        // Auto-déclenchement si HP < 25%
        if (!isSeriousModeActive && seriousCooldownTimer <= 0f && HpPercent <= 0.25f)
        {
            hitStreak = 4;
            ActivateSeriousMode();
        }
    }

    public void ReceiveDamage(float amount)
    {
        if (isDead) return;
        hitStreak = 0;
        currentHp = Mathf.Max(0f, currentHp - amount);
        animator.SetTrigger("Hit");
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, false);
        if (currentHp <= 0f) StartCoroutine(DeathRoutine());
    }

    private IEnumerator DeathRoutine()
    {
        isDead = true;
        agent.isStopped = true;
        DeactivateSeriousMode();
        animator.SetBool("IsDead", true);
        combatVfx.PlayDeath(transform.position);
        yield return new WaitForSeconds(3f);
        enabled = false;
    }

    private void TickCooldowns()
    {
        basicCooldown    = Mathf.Max(0f, basicCooldown    - Time.deltaTime);
        skillOneCooldown = Mathf.Max(0f, skillOneCooldown - Time.deltaTime);
        skillTwoCooldown = Mathf.Max(0f, skillTwoCooldown - Time.deltaTime);
        ultimateCooldown = Mathf.Max(0f, ultimateCooldown - Time.deltaTime);
    }

    private void RegenerateResources()
    {
        currentHp   = Mathf.Min(stats.maxHp,   currentHp   + stats.hpRegen   * Time.deltaTime);
        currentMana = Mathf.Min(stats.maxMana,  currentMana + stats.manaRegen * Time.deltaTime);
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, agent.velocity.sqrMagnitude > 0.01f);
    }

    private void UpdateAnimator()
    {
        Vector3 planarVel = agent.velocity; planarVel.y = 0f;
        float effectiveAS = stats.attackSpeed +
            (isSeriousModeActive ? heroDefinition.seriousAtkSpeed.Evaluate(heroLevel) : 0f);
        animator.SetFloat("MoveSpeed",   planarVel.magnitude);
        animator.SetFloat("AttackSpeed", effectiveAS);
    }

    private bool CanCast(AbilityRankData ability, int rank, float cd)
    {
        if (heroDefinition == null || isDead || cd > 0f) return false;
        return currentMana >= ability.ManaAtRank(rank);
    }

    private void SpendMana(float amount)
        => currentMana = Mathf.Max(0f, currentMana - amount);
}

// VashCoatSurface.shader
// URP shader pour le manteau rouge iconique de Vash.
// Gère : locomotion flap, Mode Sérieux rim pulse, dodge flash, dissolve mort.

Shader "Vash/CoatSurface"
{
    Properties
    {
        _BaseMap          ("Base Map", 2D)                        = "white" {}
        _BaseColor        ("Base Color", Color)                   = (0.75, 0.04, 0.04, 1)
        _SeriousRim       ("Serious Rim Color", Color)            = (1.0, 0.85, 0.2, 1)
        _MoveBlend        ("Move Blend", Range(0,1))              = 0
        _SeriousIntensity ("Serious Intensity", Range(0,1))       = 0
        _DodgeFlash       ("Dodge Flash", Range(0,1))             = 0
        _HpFade           ("HP Fade", Range(0,1))                 = 0
        _Dissolve         ("Dissolve", Range(0,1))                = 0
        _DissolveSoftness ("Dissolve Softness", Range(0.001,0.2)) = 0.04
        _NoiseScale       ("Noise Scale", Range(1,25))            = 8
        _FlapSpeed        ("Coat Flap Speed", Range(0, 10))       = 4.0
        _FlapAmplitude    ("Coat Flap Amplitude", Range(0, 0.02)) = 0.008
    }

    SubShader
    {
        Tags { "RenderType"="TransparentCutout" "Queue"="AlphaTest" "RenderPipeline"="UniversalPipeline" }

        Pass
        {
            Cull Back
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_BaseMap); SAMPLER(sampler_BaseMap);

            CBUFFER_START(UnityPerMaterial)
                float4 _BaseMap_ST;
                float4 _BaseColor;
                float4 _SeriousRim;
                float  _MoveBlend;
                float  _SeriousIntensity;
                float  _DodgeFlash;
                float  _HpFade;
                float  _Dissolve;
                float  _DissolveSoftness;
                float  _NoiseScale;
                float  _FlapSpeed;
                float  _FlapAmplitude;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float3 normalOS   : NORMAL;
                float2 uv         : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv         : TEXCOORD0;
                float3 worldPos   : TEXCOORD1;
                float3 normalWS   : TEXCOORD2;
            };

            float hash21(float2 p)
            {
                p  = frac(p * float2(123.34, 345.45));
                p += dot(p, p + 34.345);
                return frac(p.x * p.y);
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                // Claquement de manteau au déplacement — appliqué aux bas du mesh (y bas)
                float coatEdge   = saturate(-IN.positionOS.y * 0.5 + 0.5);
                float flapOffset = sin(_Time.y * _FlapSpeed + IN.positionOS.x * 5.0)
                                 * _FlapAmplitude * _MoveBlend * coatEdge;
                float3 displaced = IN.positionOS.xyz + IN.normalOS * flapOffset;
                VertexPositionInputs pos = GetVertexPositionInputs(displaced);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = TRANSFORM_TEX(IN.uv, _BaseMap);
                OUT.worldPos   = pos.positionWS;
                OUT.normalWS   = normalize(nor.normalWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                half4 albedo  = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, IN.uv) * _BaseColor;
                float noise   = hash21(floor(IN.worldPos.xz * _NoiseScale));
                float dThresh = saturate(_Dissolve * (1.0 + _DissolveSoftness));
                clip(noise - dThresh);

                float rim          = pow(1.0 - saturate(dot(normalize(IN.normalWS),
                                     float3(0,1,0))), 3.0);
                float seriousPulse = (0.5 + 0.5 * sin(_Time.y * 8.0)) * _SeriousIntensity;
                float dodgePulse   = (0.5 + 0.5 * sin(_Time.y * 25.0)) * _DodgeFlash;
                float lowHpPulse   = (0.25 + 0.75 * sin(_Time.y * 10.0)) * _HpFade;

                half3 emission = _SeriousRim.rgb * (rim * seriousPulse)
                               + half3(1, 0.95, 0.8) * dodgePulse * 0.6
                               + half3(1, 0.1, 0)    * lowHpPulse;

                return half4(albedo.rgb + emission, albedo.a);
            }
            ENDHLSL
        }
    }

// AngelArmBeam.shader
// Rayon de l'Angel Arm — destruction massive en ligne.
// Faisceau cyan-blanc avec couronne d'énergie, pulsation et bords effilochés.

Shader "Vash/AngelArmBeam"
{
    Properties
    {
        _CoreColor  ("Core Color", Color)              = (0.9, 1.0, 1.0, 1.0)
        _BeamColor  ("Beam Color", Color)              = (0.05, 0.85, 0.95, 0.85)
        _EdgeColor  ("Edge Glow Color", Color)         = (0.2, 0.6, 1.0, 0.60)
        _BeamLength ("Beam Length", Float)             = 22.0
        _BeamWidth  ("Beam Width", Range(0.1, 3))      = 1.0
        _PulseSpeed ("Pulse Speed", Range(0, 8))       = 5.0
        _NoiseFreq  ("Edge Noise Freq", Range(1, 20))  = 8.0
        _Intensity  ("Intensity", Range(0, 15))        = 6.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha One
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _CoreColor;
                float4 _BeamColor;
                float4 _EdgeColor;
                float  _BeamLength;
                float  _BeamWidth;
                float  _PulseSpeed;
                float  _NoiseFreq;
                float  _Intensity;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            float hash11(float p) { return frac(sin(p * 127.1) * 43758.5); }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = IN.uv;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                // Bords effilochés sur la longueur
                float noiseEdge = hash11(floor(IN.uv.x * _NoiseFreq + _Time.y * 2.0)) * 0.15;
                float beamEdge  = pow(saturate(1.0 - abs(IN.uv.y * 2.0 - 1.0) + noiseEdge), _BeamWidth);

                // Noyau ultra-lumineux au centre
                float core      = pow(saturate(1.0 - abs(IN.uv.y * 2.0 - 1.0)), _BeamWidth * 4.0);

                // Pulsation d'énergie le long de l'axe
                float pulse     = 0.75 + 0.25 * sin(_Time.y * _PulseSpeed * 6.2831
                                + IN.uv.x * 8.0);

                half3 color = _CoreColor.rgb  * core * pulse
                            + _BeamColor.rgb  * beamEdge * pulse
                            + _EdgeColor.rgb  * (beamEdge * 0.4);

                float alpha = saturate(beamEdge * _BeamColor.a + core * _CoreColor.a);
                return half4(color * _Intensity, alpha);
            }
