Shield System

Active energy shield overlays the armor as a translucent blue-teal forcefield. Absorbs incoming damage before HP is reduced. Shield displays as a rechargeable bar in UI. When depleted, the field collapses with a sharp electrical burst. Shield regenerates automatically after a brief delay. Shield layer visible on the hero mesh as a Fresnel-edge emission.

System Proposal

Master Chief is a frontline Tank with a unique dual HP+Shield bar system. Shield absorbs all incoming damage first; HP only depletes once shield is broken. Shield auto-regenerates after 4s without damage (delay shrinks with level).

Abilities revolve around the Gravity Hammer, providing massive crowd control: knockback, airborne lift, slow, and area denial. No mana — Chief uses a Stamina bar (regenerates continuously), keeping combat fast and aggressive.

// HeroDefinition.cs
// ScriptableObject data-driven hero definition — supports MOBA 15-level model.
// Shared across all heroes. Master Chief uses Tank role + Shield system.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string heroId = "master_chief";
    public string displayName = "Master Chief";
    public HeroRole role = HeroRole.Tank;

    [Header("Core progression — Level 1 to 15")]
    public ScaledStat maxHp = new(4800f, 9200f);
    public ScaledStat hpRegen = new(14.0f, 28.0f);
    public ScaledStat maxShield = new(600f, 1400f);
    public ScaledStat shieldRegen = new(80f, 180f);
    public ScaledStat shieldRegenDelay = new(4.0f, 2.5f);
    public ScaledStat maxStamina = new(100f, 100f);
    public ScaledStat staminaRegen = new(18f, 28f);
    public ScaledStat physicalAttack = new(160f, 310f);
    public ScaledStat physicalDefense = new(55f, 130f);
    public ScaledStat magicDefense = new(35f, 85f);
    public ScaledStat attackSpeed = new(0.72f, 0.98f);

    [Header("Fixed hero properties")]
    public float moveSpeed = 225f;
    public float basicAttackRange = 3.8f;
    public float basicAttackAngle = 120f;
    public float attackSpeedRatio = 1.0f;

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new("Hammer Swing", new[] { 0f }, new[] { 0f }, new[] { 0f }, 0f);
    public AbilityRankData skillOne = new("Gravity Slam", new[] { 260f, 330f, 400f, 470f }, new[] { 30f, 32f, 34f, 36f }, new[] { 9f, 8f, 7f, 6f }, 0f);
    public AbilityRankData skillTwo = new("Spartan Charge", new[] { 180f, 230f, 280f, 330f }, new[] { 25f, 27f, 29f, 31f }, new[] { 14f, 13f, 12f, 11f }, 0f);
    public AbilityRankData ultimate = new("Gravity Surge", new[] { 360f, 495f, 630f }, new[] { 80f, 90f, 100f }, new[] { 100f, 90f, 80f }, 0f);

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level = level,
            maxHp = maxHp.Evaluate(level),
            hpRegen = hpRegen.Evaluate(level),
            maxShield = maxShield.Evaluate(level),
            shieldRegen = shieldRegen.Evaluate(level),
            shieldRegenDelay = shieldRegenDelay.Evaluate(level),
            maxStamina = maxStamina.Evaluate(level),
            staminaRegen = staminaRegen.Evaluate(level),
            physicalAttack = physicalAttack.Evaluate(level),
            physicalDefense = physicalDefense.Evaluate(level),
            magicDefense = magicDefense.Evaluate(level),
            attackSpeed = attackSpeed.Evaluate(level),
            moveSpeed = moveSpeed,
            basicAttackRange = basicAttackRange,
            basicAttackAngle = basicAttackAngle,
            attackSpeedRatio = attackSpeedRatio
        };
    }
}

public enum HeroRole { Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin }

[Serializable]
public struct ScaledStat
{
    public float level1;
    public float level15;
    public ScaledStat(float level1, float level15) { this.level1 = level1; this.level15 = level15; }
    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        if (level == 1) return level1;
        float t = (level - 1f) / 14f;
        return Mathf.Lerp(level1, level15, t);
    }
}

[Serializable]
public class AbilityRankData
{
    public string abilityName;
    public float[] baseDamage;
    public float[] staminaCost;
    public float[] cooldown;
    public float scalingPhysical;

    public AbilityRankData(string name, float[] dmg, float[] cost, float[] cd, float scale)
    { abilityName = name; baseDamage = dmg; staminaCost = cost; cooldown = cd; scalingPhysical = scale; }

    public float DamageAtRank(int rank, float physAtk)
        => baseDamage[Mathf.Clamp(rank - 1, 0, baseDamage.Length - 1)] + physAtk * scalingPhysical;
    public float StaminaAtRank(int rank)
        => staminaCost[Mathf.Clamp(rank - 1, 0, staminaCost.Length - 1)];
    public float CooldownAtRank(int rank)
        => cooldown[Mathf.Clamp(rank - 1, 0, cooldown.Length - 1)];
}

[Serializable]
public struct RuntimeHeroStats
{
    public int level;
    public float maxHp, hpRegen;
    public float maxShield, shieldRegen, shieldRegenDelay;
    public float maxStamina, staminaRegen;
    public float physicalAttack, physicalDefense, magicDefense;
    public float attackSpeed, moveSpeed, basicAttackRange, basicAttackAngle, attackSpeedRatio;
}
C# MasterChiefController.cs
// MasterChiefController.cs
// Main hero controller: movement, dual HP+Shield system, stamina, ability casts, level-up.
// Shield absorbs ALL incoming damage first. HP only depletes when shield is broken.
// Shield auto-regenerates after shieldRegenDelay seconds without taking damage.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class MasterChiefController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator animator;
    [SerializeField] private NavMeshAgent agent;
    [SerializeField] private ChiefCombatVFX combatVfx;
    [SerializeField] private Transform hammerSpawn;
    [SerializeField] private LayerMask groundMask;
    [SerializeField] private LayerMask enemyMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentShield;
    [SerializeField] private float currentStamina;
    [SerializeField] private bool isDead;

    private float shieldRegenTimer;
    private float basicCooldown, skillOneCooldown, skillTwoCooldown, ultimateCooldown;
    private bool isUltimateActive;
    private float ultimateOverchargeShield;

    public float HpPercent    => stats.maxHp    <= 0f ? 0f : currentHp / stats.maxHp;
    public float ShieldPercent => stats.maxShield <= 0f ? 0f : currentShield / stats.maxShield;
    public float StaminaPercent => stats.maxStamina <= 0f ? 0f : currentStamina / stats.maxStamina;
    public bool  IsDead => isDead;

    private void Reset()
    {
        agent    = GetComponent<NavMeshAgent>();
        animator = GetComponentInChildren<Animator>();
    }

    private void Awake() { if (agent == null) agent = GetComponent<NavMeshAgent>(); ApplyLevel(heroLevel, true); }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        RegenerateResources();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore) { currentHp = stats.maxHp; currentShield = stats.maxShield; currentStamina = stats.maxStamina; }
        else { currentHp = Mathf.Min(currentHp, stats.maxHp); currentShield = Mathf.Min(currentShield, stats.maxShield); }
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
        animator.SetBool("IsMoving", true);
    }

    // ── BASIC ATTACK: wide frontal AoE cleave ──────────────────────────────
    public void BasicAttack(Vector3 direction)
    {
        if (isDead || basicCooldown > 0f) return;
        basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
        animator.SetTrigger("BasicAttack");
        float dmg = stats.physicalAttack * 1.15f;
        ApplyAoeCleave(transform.position, direction, stats.basicAttackRange, stats.basicAttackAngle, dmg);
        combatVfx.PlayHammerSwing(hammerSpawn.position);
    }

    // ── SKILL 1: Gravity Slam — radial knockback + airborne ───────────────
    public bool TryCastSkillOne(Vector3 groundTarget)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;
        SpendStamina(heroDefinition.skillOne.StaminaAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        agent.isStopped = true;
        transform.LookAt(new Vector3(groundTarget.x, transform.position.y, groundTarget.z));
        animator.SetTrigger("CastSkill01");
        StartCoroutine(GravitySlamRoutine(groundTarget, skillOneRank));
        return true;
    }

    private IEnumerator GravitySlamRoutine(Vector3 center, int rank)
    {
        yield return new WaitForSeconds(0.35f); // wind-up delay
        float radius = 6f;
        float damage = heroDefinition.skillOne.DamageAtRank(rank, stats.physicalAttack);
        float knockbackDist = 3.5f + rank * 0.5f;
        float airborneDuration = 0.5f + rank * 0.1f;
        combatVfx.PlayGravitySlam(center, radius);
        foreach (var col in Physics.OverlapSphere(center, radius, enemyMask))
        {
            var health = col.GetComponent<Health>();
            if (health != null) health.ApplyPhysicalDamage(damage);
            var cc = col.GetComponent<CrowdControlReceiver>();
            if (cc != null)
            {
                Vector3 pushDir = (col.transform.position - center).normalized;
                cc.ApplyKnockback(pushDir, knockbackDist);
                cc.ApplyAirborne(airborneDuration);
            }
        }
        agent.isStopped = false;
    }

    // ── SKILL 2: Spartan Charge — dash + slow on impact ───────────────────
    public bool TryCastSkillTwo(Vector3 direction)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;
        SpendStamina(heroDefinition.skillTwo.StaminaAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);
        StartCoroutine(SpartanChargeRoutine(direction, skillTwoRank));
        return true;
    }

    private IEnumerator SpartanChargeRoutine(Vector3 dir, int rank)
    {
        agent.isStopped = true;
        animator.SetTrigger("Charge");
        combatVfx.PlayChargeStart(transform.position);
        float dashRange = 8f;
        float speed = 18f;
        float elapsed = 0f;
        float duration = dashRange / speed;
        dir = new Vector3(dir.x, 0, dir.z).normalized;
        bool hit = false;
        while (elapsed < duration && !hit)
        {
            transform.Translate(dir * speed * Time.deltaTime, Space.World);
            foreach (var col in Physics.OverlapSphere(transform.position, 1.5f, enemyMask))
            {
                var health = col.GetComponent<Health>();
                if (health != null)
                {
                    health.ApplyPhysicalDamage(heroDefinition.skillTwo.DamageAtRank(rank, stats.physicalAttack));
                    var cc = col.GetComponent<CrowdControlReceiver>();
                    if (cc != null) cc.ApplySlow(0.5f, 1.5f + rank * 0.25f);
                    hit = true;
                }
            }
            elapsed += Time.deltaTime;
            yield return null;
        }
        agent.Warp(transform.position);
        agent.isStopped = false;
        combatVfx.PlayChargeEnd(transform.position);
    }

    // ── ULTIMATE: Gravity Surge — lift field + shield overcharge ──────────
    public bool TryCastUltimate()
    {
        if (!CanCast(heroDefinition.ultimate, ultimateRank, ultimateCooldown)) return false;
        SpendStamina(heroDefinition.ultimate.StaminaAtRank(ultimateRank));
        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        StartCoroutine(GravitySurgeRoutine(ultimateRank));
        return true;
    }

    private IEnumerator GravitySurgeRoutine(int rank)
    {
        isUltimateActive = true;
        agent.isStopped = true;
        animator.SetTrigger("CastUltimate");
        float duration = 2f + rank; // 3s / 4s / 5s
        float radius = 12f + rank * 2f;
        float dmgPerTick = heroDefinition.ultimate.DamageAtRank(rank, stats.physicalAttack) / (duration * 2);
        float shieldMultiplier = 1.1f + rank * 0.1f;
        ultimateOverchargeShield = stats.maxShield * shieldMultiplier;
        currentShield = ultimateOverchargeShield;
        combatVfx.PlayGravitySurge(transform.position, radius, duration);
        float elapsed = 0f;
        float tickTimer = 0f;
        while (elapsed < duration)
        {
            tickTimer += Time.deltaTime;
            if (tickTimer >= 0.5f)
            {
                tickTimer = 0f;
                foreach (var col in Physics.OverlapSphere(transform.position, radius, enemyMask))
                {
                    col.GetComponent<Health>()?.ApplyPhysicalDamage(dmgPerTick);
                    col.GetComponent<CrowdControlReceiver>()?.ApplyAirborne(0.6f);
                }
            }
            elapsed += Time.deltaTime;
            yield return null;
        }
        isUltimateActive = false;
        agent.isStopped = false;
    }

    // ── DAMAGE RECEPTION: Shield absorbs first ────────────────────────────
    public void ReceiveDamage(float amount)
    {
        if (isDead) return;
        shieldRegenTimer = stats.shieldRegenDelay; // reset regen delay on any hit
        if (currentShield > 0f)
        {
            float absorbed = Mathf.Min(currentShield, amount);
            currentShield -= absorbed;
            amount -= absorbed;
            combatVfx.PlayShieldHit(transform.position, currentShield / stats.maxShield);
            if (currentShield <= 0f) combatVfx.PlayShieldBreak(transform.position);
        }
        if (amount > 0f)
        {
            currentHp = Mathf.Max(0f, currentHp - amount);
            animator.SetTrigger("Hit");
        }
        combatVfx.UpdateHeroState(HpPercent, ShieldPercent, false);
        if (currentHp <= 0f) StartCoroutine(DeathRoutine());
    }

    private IEnumerator DeathRoutine()
    {
        isDead = true;
        agent.isStopped = true;
        animator.SetBool("IsDead", true);
        combatVfx.PlayDeath(transform.position);
        yield return new WaitForSeconds(3f);
        enabled = false;
    }

    private void TickCooldowns()
    {
        basicCooldown    = Mathf.Max(0f, basicCooldown - Time.deltaTime);
        skillOneCooldown = Mathf.Max(0f, skillOneCooldown - Time.deltaTime);
        skillTwoCooldown = Mathf.Max(0f, skillTwoCooldown - Time.deltaTime);
        ultimateCooldown = Mathf.Max(0f, ultimateCooldown - Time.deltaTime);
    }

    private void RegenerateResources()
    {
        // HP regeneration
        currentHp = Mathf.Min(stats.maxHp, currentHp + stats.hpRegen * Time.deltaTime);
        // Stamina regeneration (continuous)
        currentStamina = Mathf.Min(stats.maxStamina, currentStamina + stats.staminaRegen * Time.deltaTime);
        // Shield regeneration (delayed, disabled during ultimate)
        if (!isUltimateActive)
        {
            shieldRegenTimer = Mathf.Max(0f, shieldRegenTimer - Time.deltaTime);
            if (shieldRegenTimer <= 0f)
                currentShield = Mathf.Min(stats.maxShield, currentShield + stats.shieldRegen * Time.deltaTime);
        }
        combatVfx.UpdateHeroState(HpPercent, ShieldPercent, agent.velocity.sqrMagnitude > 0.01f);
    }

    private void UpdateAnimator()
    {
        Vector3 vel = agent.velocity; vel.y = 0f;
        animator.SetFloat("MoveSpeed", vel.magnitude);
        animator.SetFloat("AttackSpeed", stats.attackSpeed);
        animator.SetFloat("ShieldPercent", ShieldPercent);
    }

    private void ApplyAoeCleave(Vector3 origin, Vector3 forward, float radius, float angleDeg, float damage)
    {
        foreach (var col in Physics.OverlapSphere(origin, radius, enemyMask))
        {
            Vector3 toEnemy = (col.transform.position - origin).normalized;
            if (Vector3.Angle(forward, toEnemy) <= angleDeg * 0.5f)
                col.GetComponent<Health>()?.ApplyPhysicalDamage(damage);
        }
    }

    private bool CanCast(AbilityRankData ability, int rank, float cd)
        => heroDefinition != null && !isDead && cd <= 0f && currentStamina >= ability.StaminaAtRank(rank);

    private void SpendStamina(float amount)
        => currentStamina = Mathf.Max(0f, currentStamina - amount);
}

public class CrowdControlReceiver : MonoBehaviour
{
    public void ApplyKnockback(Vector3 dir, float dist) { /* push via NavMeshAgent or Rigidbody */ }
    public void ApplyAirborne(float duration)             { /* interrupt movement, play airborne anim */ }
    public void ApplySlow(float factor, float duration)   { /* reduce NavMeshAgent.speed by factor */ }
}

public class Health : MonoBehaviour
{
    [SerializeField] private float hp = 1000f;
    public void ApplyPhysicalDamage(float damage) => hp -= damage;
    public void ApplyMagicDamage(float damage)    => hp -= damage;
}
C# ChiefCombatVFX.cs
// ChiefCombatVFX.cs
// VFX binder — connects gameplay events to materials, shaders, and particle systems.
// Drives shield glow intensity from shield percent, hammer slam shockwave, gravity surge field.

using UnityEngine;

public class ChiefCombatVFX : MonoBehaviour
{
    [Header("Particle Systems")]
    [SerializeField] private ParticleSystem hammerSwingFx;
    [SerializeField] private ParticleSystem gravitySlamFx;
    [SerializeField] private ParticleSystem chargeTrailFx;
    [SerializeField] private ParticleSystem gravitySurgeFx;
    [SerializeField] private ParticleSystem shieldHitFx;

// ChiefArmorSurface.shader
// Gameplay-aware armor shader: HP damage pulse (orange glow at low HP),
// movement micro-vibration, shield integration hint, dissolve on death.

Shader "MasterChief/ArmorSurface"
{
    Properties
    {
        _BaseMap       ("Base Map", 2D)              = "white" {}
        _BaseColor     ("Base Color", Color)          = (0.18, 0.29, 0.17, 1)
        _EmissionColor ("Damage Pulse Color", Color)  = (1.0, 0.44, 0.0, 1)
        _HpFade        ("HP Fade", Range(0,1))        = 0
        _MoveBlend     ("Move Blend", Range(0,1))     = 0
        _Dissolve      ("Dissolve", Range(0,1))       = 0
        _DissolveSoft  ("Dissolve Softness", Range(0.001,0.2)) = 0.05
        _NoiseScale    ("Noise Scale", Range(1,25))   = 10
    }

    SubShader
    {
        Tags { "RenderType"="TransparentCutout" "Queue"="AlphaTest" "RenderPipeline"="UniversalPipeline" }

        Pass
        {
            Cull Back
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_BaseMap); SAMPLER(sampler_BaseMap);

            CBUFFER_START(UnityPerMaterial)
                float4 _BaseMap_ST;
                float4 _BaseColor;
                float4 _EmissionColor;
                float  _HpFade;
                float  _MoveBlend;
                float  _Dissolve;
                float  _DissolveSoft;
                float  _NoiseScale;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float3 normalOS : NORMAL; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; float3 worldPos : TEXCOORD1; float3 normalWS : TEXCOORD2; };

            float hash21(float2 p)
            {
                p = frac(p * float2(127.1, 311.7));
                p += dot(p, p + 34.345);
                return frac(p.x * p.y);
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                // Micro-vibration on move (armor hum)
                float moveShake = sin(_Time.y * 14.0 + IN.positionOS.y * 8.0) * 0.002 * _MoveBlend;
                float3 displaced = IN.positionOS.xyz + IN.normalOS * moveShake;
                VertexPositionInputs pos = GetVertexPositionInputs(displaced);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = TRANSFORM_TEX(IN.uv, _BaseMap);
                OUT.worldPos   = pos.positionWS;
                OUT.normalWS   = normalize(nor.normalWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                // Death dissolve
                float noise = hash21(floor(IN.worldPos.xz * _NoiseScale));
                clip(noise - saturate(_Dissolve * (1.0 + _DissolveSoft)));

                half4 albedo = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, IN.uv) * _BaseColor;

                // Rim edge from normals
                float rim = pow(1.0 - saturate(IN.normalWS.y), 2.0);
                // Low HP orange pulse
                float damagePulse = (0.3 + 0.7 * sin(_Time.y * 9.0)) * _HpFade;
                half3 emission = _EmissionColor.rgb * (rim * damagePulse * 1.8 + damagePulse * 0.4);

                return half4(albedo.rgb + emission, albedo.a);
            }
            ENDHLSL
        }
    }

// MjolnirShieldField.shader
// Transparent energy shield overlay for the hero mesh.
// Driven by _ShieldGlow (0=broken, 1=full). Fresnel edge glow, hexagonal pattern, hit ripple.

Shader "MasterChief/ShieldField"
{
    Properties
    {
        _ShieldColor  ("Shield Color",  Color)           = (0.18, 0.82, 0.98, 0.18)
        _EdgeColor    ("Edge Color",    Color)           = (0.4, 0.95, 1.0, 1.0)
        _ShieldGlow   ("Shield Glow",   Range(0,1))      = 1.0
        _FresnelPow   ("Fresnel Power", Range(1,8))      = 3.5
        _HexScale     ("Hex Scale",     Range(2,30))     = 12.0
        _HexIntensity ("Hex Intensity", Range(0,1))      = 0.18
        _PulseSpeed   ("Pulse Speed",   Range(0,6))      = 2.2
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent+10" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Back

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _ShieldColor;
                float4 _EdgeColor;
                float  _ShieldGlow;
                float  _FresnelPow;
                float  _HexScale;
                float  _HexIntensity;
                float  _PulseSpeed;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float3 normalOS : NORMAL; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float3 normalWS : TEXCOORD0; float3 viewDirWS : TEXCOORD1; float2 uv : TEXCOORD2; };

            // Approximate hexagonal grid pattern
            float hexGrid(float2 uv, float scale)
            {
                uv *= scale;
                float2 r = float2(1.0, 1.732);
                float2 h = r * 0.5;
                float2 a = fmod(uv, r) - h;
                float2 b = fmod(uv - h, r) - h;
                float2 gv = dot(a, a) < dot(b, b) ? a : b;
                return smoothstep(0.42, 0.48, length(gv));
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.normalWS   = normalize(nor.normalWS);
                OUT.viewDirWS  = GetWorldSpaceNormalizeViewDir(pos.positionWS);
                OUT.uv         = IN.uv;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                // Shield Fresnel rim
                float fresnel = pow(1.0 - saturate(dot(IN.normalWS, IN.viewDirWS)), _FresnelPow);

                // Hexagonal surface pattern
                float hex = hexGrid(IN.uv, _HexScale) * _HexIntensity;

                // Pulse animation
                float pulse = 0.88 + 0.12 * sin(_Time.y * _PulseSpeed * 6.2831);

                float shieldAlpha = saturate((fresnel * 0.7 + hex + 0.05) * _ShieldGlow * pulse);
                half3 color = lerp(_ShieldColor.rgb, _EdgeColor.rgb, fresnel);

                return half4(color, shieldAlpha * _ShieldColor.a * _ShieldGlow);
            }
            ENDHLSL
        }
    }
}
