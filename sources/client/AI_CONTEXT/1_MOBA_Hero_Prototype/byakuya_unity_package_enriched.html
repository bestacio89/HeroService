// HeroDefinition.cs
// ScriptableObject data-driven pour respecter un modèle MOBA 15 niveaux
// Chaque héros possède ses propres stats de base, croissance et rangs de compétences.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string heroId = "byakuya";
    public string displayName = "Byakuya Kuchiki";
    public HeroRole role = HeroRole.MageAssassin;

    [Header("Core progression - Niveau 1 à 15")]
    public ScaledStat maxHp = new(2580f, 5270f);
    public ScaledStat hpRegen = new(7.2f, 15.6f);
    public ScaledStat maxMana = new(500f, 1900f);
    public ScaledStat manaRegen = new(4.0f, 6.8f);
    public ScaledStat physicalAttack = new(121f, 247f);
    public ScaledStat magicPower = new(0f, 0f);
    public ScaledStat physicalDefense = new(18f, 60f);
    public ScaledStat magicDefense = new(15f, 50f);
    public ScaledStat attackSpeed = new(1.03f, 1.31f);

    [Header("Fixed hero properties")]
    public float moveSpeed = 255f;
    public float basicAttackRange = 3.2f;
    public float attackSpeedRatio = 1.0f;
    public float criticalDamage = 2.0f;

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new("Basic", new[] { 75f }, new[] { 0f }, new[] { 0f }, 0f);
    public AbilityRankData skillOne = new("Senbonzakura Shot", new[] { 110f, 150f, 190f, 230f }, new[] { 45f, 50f, 55f, 60f }, new[] { 8f, 7.5f, 7f, 6.5f }, 0.70f);
    public AbilityRankData skillTwo = new("Shunpo Blossom", new[] { 90f, 130f, 170f, 210f }, new[] { 60f, 65f, 70f, 75f }, new[] { 13f, 12f, 11f, 10f }, 0.55f);
    public AbilityRankData ultimate = new("Senkei: Bankai", new[] { 260f, 360f, 460f }, new[] { 120f, 140f, 160f }, new[] { 42f, 36f, 30f }, 1.05f);

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level = level,
            maxHp = maxHp.Evaluate(level),
            hpRegen = hpRegen.Evaluate(level),
            maxMana = maxMana.Evaluate(level),
            manaRegen = manaRegen.Evaluate(level),
            physicalAttack = physicalAttack.Evaluate(level),
            magicPower = magicPower.Evaluate(level),
            physicalDefense = physicalDefense.Evaluate(level),
            magicDefense = magicDefense.Evaluate(level),
            attackSpeed = attackSpeed.Evaluate(level),
            moveSpeed = moveSpeed,
            basicAttackRange = basicAttackRange,
            attackSpeedRatio = attackSpeedRatio,
            criticalDamage = criticalDamage
        };
    }
}

[Serializable]
public struct ScaledStat
{
    public float level1;
    public float level15;

    public ScaledStat(float level1, float level15)
    {
        this.level1 = level1;
        this.level15 = level15;
    }

    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        if (level == 1) return level1;
        float t = (level - 1f) / 14f;
        return Mathf.Lerp(level1, level15, t);
    }
}

[Serializable]
public class AbilityRankData
{
    public string abilityName;
    public float[] baseDamage;
    public float[] manaCost;
    public float[] cooldown;
    public float scalingMagicPower;

    public AbilityRankData(string abilityName, float[] baseDamage, float[] manaCost, float[] cooldown, float scalingMagicPower)
    {
        this.abilityName = abilityName;
        this.baseDamage = baseDamage;
        this.manaCost = manaCost;
        this.cooldown = cooldown;
        this.scalingMagicPower = scalingMagicPower;
    }

    public float DamageAtRank(int rank, float magicPower)
        => baseDamage[Mathf.Clamp(rank - 1, 0, baseDamage.Length - 1)] + magicPower * scalingMagicPower;

    public float ManaAtRank(int rank)
        => manaCost[Mathf.Clamp(rank - 1, 0, manaCost.Length - 1)];

    public float CooldownAtRank(int rank)
        => cooldown[Mathf.Clamp(rank - 1, 0, cooldown.Length - 1)];
}

    public float basicAttackRange;
    public float attackSpeedRatio;
    public float criticalDamage;
}
C# ByakuyaHeroController.cs
// ByakuyaHeroController.cs
// Contrôleur principal : mouvement, montée en niveau, régénération, cast et liens Animator/VFX.
// Découpage conseillé Unity :
// - HeroDefinition.asset pour les données
// - ByakuyaHeroController sur le prefab du héros
// - CharacterController ou NavMeshAgent selon votre mode de déplacement
// - Animator sur le mesh 3D importé depuis Meshy

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class ByakuyaHeroController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator animator;
    [SerializeField] private NavMeshAgent agent;
    [SerializeField] private ByakuyaCombatVFX combatVfx;
    [SerializeField] private Transform skillSpawn;
    [SerializeField] private LayerMask groundMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentMana;
    [SerializeField] private bool isDead;

    private float basicCooldown;
    private float skillOneCooldown;
    private float skillTwoCooldown;
    private float ultimateCooldown;

    public RuntimeHeroStats Stats => stats;
    public float HpPercent => stats.maxHp <= 0f ? 0f : currentHp / stats.maxHp;
    public float ManaPercent => stats.maxMana <= 0f ? 0f : currentMana / stats.maxMana;
    public bool IsDead => isDead;

    private void Reset()
    {
        agent = GetComponent<NavMeshAgent>();
        animator = GetComponentInChildren<Animator>();
    }

    private void Awake()
    {
        if (agent == null) agent = GetComponent<NavMeshAgent>();
        ApplyLevel(heroLevel, true);
    }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;

        TickCooldowns();
        RegenerateResources();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats = heroDefinition.CreateRuntime(heroLevel);

        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore)
        {
            currentHp = stats.maxHp;
            currentMana = stats.maxMana;
        }
        else
        {
            currentHp = Mathf.Min(currentHp, stats.maxHp);
            currentMana = Mathf.Min(currentMana, stats.maxMana);
        }
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
        animator.SetBool("IsMoving", true);
    }

    public bool TryCastSkillOne(Transform target)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;

        SpendMana(heroDefinition.skillOne.ManaAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        agent.isStopped = true;
        transform.LookAt(new Vector3(target.position.x, transform.position.y, target.position.z));
        animator.SetTrigger("CastSkill01");
        combatVfx.PlayPetalCast(skillSpawn.position, target.position);
        return true;
    }

    public bool TryCastSkillTwo(Vector3 destination)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;

        SpendMana(heroDefinition.skillTwo.ManaAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);
        StartCoroutine(ShunpoRoutine(destination));
        return true;
    }

    public bool TryCastUltimate(Vector3 center)
    {
        if (!CanCast(heroDefinition.ultimate, ultimateRank, ultimateCooldown)) return false;

        SpendMana(heroDefinition.ultimate.ManaAtRank(ultimateRank));
        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        agent.isStopped = true;
        animator.SetTrigger("CastUltimate");
        combatVfx.PlayBankai(center, 5f + ultimateRank * 1.5f);
        return true;
    }

    public void BasicAttack(Health target)
    {
        if (target == null || basicCooldown > 0f || isDead) return;

        float damage = heroDefinition.basicAttack.DamageAtRank(1, stats.magicPower) + stats.physicalAttack;
        basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
        animator.SetTrigger("BasicAttack");
        target.ApplyPhysicalDamage(damage);
    }

    public void ReceiveDamage(float amount)
    {
        if (isDead) return;
        currentHp = Mathf.Max(0f, currentHp - amount);
        animator.SetTrigger("Hit");
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, false);

        if (currentHp <= 0f)
        {
            StartCoroutine(DeathRoutine());
        }
    }

    private IEnumerator ShunpoRoutine(Vector3 destination)
    {
        agent.isStopped = true;
        animator.SetTrigger("Teleport");
        combatVfx.PlayTeleportStart(transform.position);
        yield return new WaitForSeconds(0.08f);
        transform.position = destination;
        combatVfx.PlayTeleportEnd(transform.position);
        yield return new WaitForSeconds(0.05f);
        agent.Warp(destination);
        agent.isStopped = false;
    }

    private IEnumerator DeathRoutine()
    {
        isDead = true;
        agent.isStopped = true;
        animator.SetBool("IsDead", true);
        combatVfx.PlayDeath(transform.position);
        yield return new WaitForSeconds(3f);
        enabled = false;
    }

    private void TickCooldowns()
    {
        basicCooldown = Mathf.Max(0f, basicCooldown - Time.deltaTime);
        skillOneCooldown = Mathf.Max(0f, skillOneCooldown - Time.deltaTime);
        skillTwoCooldown = Mathf.Max(0f, skillTwoCooldown - Time.deltaTime);
        ultimateCooldown = Mathf.Max(0f, ultimateCooldown - Time.deltaTime);
    }

    private void RegenerateResources()
    {
        currentHp = Mathf.Min(stats.maxHp, currentHp + stats.hpRegen * Time.deltaTime);
        currentMana = Mathf.Min(stats.maxMana, currentMana + stats.manaRegen * Time.deltaTime);
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, agent.velocity.sqrMagnitude > 0.01f);
    }

    private void UpdateAnimator()
    {
        Vector3 planarVelocity = agent.velocity;
        planarVelocity.y = 0f;
        animator.SetFloat("MoveSpeed", planarVelocity.magnitude);
        animator.SetFloat("AttackSpeed", stats.attackSpeed);
    }

    private bool CanCast(AbilityRankData ability, int rank, float currentCooldown)
    {
        if (heroDefinition == null || isDead || currentCooldown > 0f) return false;
        return currentMana >= ability.ManaAtRank(rank);
    }

    private void SpendMana(float amount)
    {
        currentMana = Mathf.Max(0f, currentMana - amount);
    }
}

public class Health : MonoBehaviour
{
    [SerializeField] private float hp = 1000f;

    public void ApplyPhysicalDamage(float damage) => hp -= damage;
    public void ApplyMagicDamage(float damage) => hp -= damage;
}
C# ByakuyaCombatVFX.cs
// ByakuyaCombatVFX.cs
// Binder simple entre gameplay et matériaux/shaders pour déplacement, téléportation, mort et cast.
// Important : ce script permet de tirer parti des shaders HLSL de l'onglet Shader sans coder la logique VFX dans le contrôleur.

using UnityEngine;

public class ByakuyaCombatVFX : MonoBehaviour
{
    [SerializeField] private ParticleSystem petalCast;
    [SerializeField] private ParticleSystem teleportBurst;
    [SerializeField] private ParticleSystem deathPetals;
    [SerializeField] private ParticleSystem bankaiGroundFx;

    [Header("Materials using HLSL shaders")]
    [SerializeField] private Renderer heroRenderer;
    [SerializeField] private Renderer teleportTrailRenderer;
    [SerializeField] private Renderer bankaiGroundRenderer;

    private static readonly int MoveBlend = Shader.PropertyToID("_MoveBlend");
    private static readonly int HpFade = Shader.PropertyToID("_HpFade");
    private static readonly int ManaGlow = Shader.PropertyToID("_ManaGlow");
    private static readonly int Dissolve = Shader.PropertyToID("_Dissolve");
    private static readonly int Radius = Shader.PropertyToID("_Radius");

    public void UpdateHeroState(float hpPercent, float manaPercent, bool isMoving)
    {
        if (heroRenderer != null)
        {
            heroRenderer.material.SetFloat(MoveBlend, isMoving ? 1f : 0f);
            heroRenderer.material.SetFloat(HpFade, 1f - hpPercent);
            heroRenderer.material.SetFloat(ManaGlow, manaPercent);
        }
    }

    public void PlayPetalCast(Vector3 origin, Vector3 target)
    {
        if (petalCast == null) return;
        petalCast.transform.position = origin;
        petalCast.transform.forward = (target - origin).normalized;
        petalCast.Play();
    }

    public void PlayTeleportStart(Vector3 position)
    {
        if (teleportBurst != null)
        {
            teleportBurst.transform.position = position;
            teleportBurst.Play();
        }
        if (teleportTrailRenderer != null)
            teleportTrailRenderer.material.SetFloat(Dissolve, 0f);
    }

    public void PlayTeleportEnd(Vector3 position)
    {
        if (teleportBurst != null)
        {
            teleportBurst.transform.position = position;
            teleportBurst.Play();
        }
        if (teleportTrailRenderer != null)
            teleportTrailRenderer.material.SetFloat(Dissolve, 0.35f);
    }

    public void PlayBankai(Vector3 center, float radius)
    {
        if (bankaiGroundFx != null)
        {
            bankaiGroundFx.transform.position = center;
            bankaiGroundFx.Play();
        }
        if (bankaiGroundRenderer != null)
            bankaiGroundRenderer.material.SetFloat(Radius, radius);
    }

    public void PlayDeath(Vector3 position)
    {
        if (deathPetals != null)
        {
            deathPetals.transform.position = position;
            deathPetals.Play();
        }
        if (heroRenderer != null)
            heroRenderer.material.SetFloat(Dissolve, 1f);
    }
}

        _Distortion ("UV Distortion", Range(0, 0.2)) = 0.035
        _FresnelPower ("Fresnel Power", Range(1, 8)) = 3.2
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_MainTex); SAMPLER(sampler_MainTex);

            CBUFFER_START(UnityPerMaterial)
                float4 _MainTex_ST;
                float4 _PetalColor;
                float4 _GlowColor;
                float _GlowIntensity;
                float _FlowSpeed;
                float _Distortion;
                float _FresnelPower;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float2 uv : TEXCOORD0;
                float3 normalOS : NORMAL;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv : TEXCOORD0;
                float3 normalWS : TEXCOORD1;
                float3 viewDirWS : TEXCOORD2;
            };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                VertexNormalInputs nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv = TRANSFORM_TEX(IN.uv, _MainTex);
                OUT.normalWS = normalize(nor.normalWS);
                OUT.viewDirWS = GetWorldSpaceNormalizeViewDir(pos.positionWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float2 flowUv = IN.uv;
                flowUv.x += sin(_Time.y * _FlowSpeed + IN.uv.y * 12.0) * _Distortion;
                flowUv.y += cos(_Time.y * _FlowSpeed * 0.7 + IN.uv.x * 10.0) * _Distortion;

                half4 tex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, flowUv);
                float fresnel = pow(1.0 - saturate(dot(normalize(IN.normalWS), normalize(IN.viewDirWS))), _FresnelPower);
                half3 emission = _GlowColor.rgb * (_GlowIntensity * tex.a + fresnel * 0.65);

                half4 col;
                col.rgb = tex.rgb * _PetalColor.rgb + emission;
                col.a = tex.a * _PetalColor.a;
                return col;
            }
            ENDHLSL
        }
    }

HLSL ByakuyaHeroSurface.shader
// ByakuyaHeroSurface.shader
// Shader héros gameplay-aware : locomotion, mana glow, hit/damage, dissolve mort.
// À appliquer au mesh principal du héros Unity pour relier animation + gameplay + VFX.

Shader "Byakuya/HeroSurface"
{
    Properties
    {
        _BaseMap ("Base Map", 2D) = "white" {}
        _BaseColor ("Base Color", Color) = (1,1,1,1)
        _EmissionColor ("Emission Color", Color) = (0.75, 0.7, 1.0, 1)
        _ManaGlow ("Mana Glow", Range(0,1)) = 1
        _MoveBlend ("Move Blend", Range(0,1)) = 0
        _HpFade ("HP Fade", Range(0,1)) = 0
        _Dissolve ("Dissolve", Range(0,1)) = 0
        _DissolveSoftness ("Dissolve Softness", Range(0.001,0.2)) = 0.04
        _NoiseScale ("Noise Scale", Range(1,25)) = 8
    }

    SubShader
    {
        Tags { "RenderType"="TransparentCutout" "Queue"="AlphaTest" "RenderPipeline"="UniversalPipeline" }

        Pass
        {
            Cull Back
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_BaseMap); SAMPLER(sampler_BaseMap);

            CBUFFER_START(UnityPerMaterial)
                float4 _BaseMap_ST;
                float4 _BaseColor;
                float4 _EmissionColor;
                float _ManaGlow;
                float _MoveBlend;
                float _HpFade;
                float _Dissolve;
                float _DissolveSoftness;
                float _NoiseScale;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float3 normalOS : NORMAL;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv : TEXCOORD0;
                float3 worldPos : TEXCOORD1;
                float3 normalWS : TEXCOORD2;
            };

            float hash21(float2 p)
            {
                p = frac(p * float2(123.34, 345.45));
                p += dot(p, p + 34.345);
                return frac(p.x * p.y);
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                float moveOffset = sin(_Time.y * 9.0 + IN.positionOS.y * 6.0) * 0.003 * _MoveBlend;
                float3 displaced = IN.positionOS.xyz + IN.normalOS * moveOffset;
                VertexPositionInputs pos = GetVertexPositionInputs(displaced);
                VertexNormalInputs nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv = TRANSFORM_TEX(IN.uv, _BaseMap);
                OUT.worldPos = pos.positionWS;
                OUT.normalWS = normalize(nor.normalWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                half4 albedo = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, IN.uv) * _BaseColor;
                float noise = hash21(floor(IN.worldPos.xz * _NoiseScale));
                float dissolveThreshold = saturate(_Dissolve * (1.0 + _DissolveSoftness));
                clip(noise - dissolveThreshold);

                float rim = pow(1.0 - saturate(IN.normalWS.y), 2.5);
                float manaPulse = (0.45 + 0.55 * sin(_Time.y * 5.0)) * _ManaGlow;
                float lowHpPulse = (0.25 + 0.75 * sin(_Time.y * 8.0)) * _HpFade;
                half3 emission = _EmissionColor.rgb * (rim * manaPulse + lowHpPulse);

                return half4(albedo.rgb + emission, albedo.a);
            }
            ENDHLSL
        }
    }
}
HLSL ByakuyaTeleportTrail.shader
// ByakuyaTeleportTrail.shader
// Effet destiné aux dashs / flash-step / téléportation / respawn.
// Utilisable sur un mesh trail, un quad ou une ribbon VFX.

Shader "Byakuya/TeleportTrail"
{
    Properties
    {
        _MainColor ("Main Color", Color) = (0.85, 0.75, 1.0, 0.85)
        _EdgeColor ("Edge Color", Color) = (1.0, 1.0, 1.0, 1.0)
        _ScrollSpeed ("Scroll Speed", Range(0, 8)) = 3.5
        _Dissolve ("Dissolve", Range(0, 1)) = 0
        _Intensity ("Intensity", Range(0, 8)) = 2.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha One
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _MainColor;
                float4 _EdgeColor;
                float _ScrollSpeed;
                float _Dissolve;
                float _Intensity;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv : TEXCOORD0;
            };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                OUT.positionCS = pos.positionCS;
                OUT.uv = IN.uv;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float scan = frac(IN.uv.x - _Time.y * _ScrollSpeed);
                float band = smoothstep(0.0, 0.2, scan) * (1.0 - smoothstep(0.2, 0.8, scan));
                float edge = pow(saturate(1.0 - abs(IN.uv.y * 2.0 - 1.0)), 4.0);
                float alpha = saturate((1.0 - _Dissolve) * (edge + band));
                half3 color = (_MainColor.rgb * edge + _EdgeColor.rgb * band) * _Intensity;
                return half4(color, alpha * _MainColor.a);
            }
            ENDHLSL
        }
    }

HLSL BankaiGround.shader
// BankaiGround.shader
// Zone ultime au sol : pulsation circulaire lisible en jeu et facile à régler dans l'Inspector.

Shader "Byakuya/BankaiGroundEffect"
{
    Properties
    {
        _Color ("Zone Color", Color) = (0.55, 0.2, 0.85, 0.35)
        _EdgeColor ("Edge Color", Color) = (0.95, 0.8, 1.0, 0.95)
        _Radius ("Radius", Float) = 8.0
        _RingWidth ("Ring Width", Range(0.05, 3)) = 0.45
        _PulseSpeed ("Pulse Speed", Range(0, 5)) = 1.8
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _EdgeColor;
                float _Radius;
                float _RingWidth;
                float _PulseSpeed;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                OUT.positionCS = pos.positionCS;
                OUT.uv = IN.uv * 2.0 - 1.0;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float dist = length(IN.uv) * _Radius;
                float pulse = 0.92 + 0.08 * sin(_Time.y * _PulseSpeed * 6.2831);
                float ring = smoothstep(_Radius, _Radius - _RingWidth, dist * pulse);
                float edge = smoothstep(_Radius, _Radius - (_RingWidth * 0.35), dist * pulse);
                half3 color = lerp(_EdgeColor.rgb, _Color.rgb, ring);
                float alpha = saturate(ring * _Color.a + edge * _EdgeColor.a * 0.45);
                return half4(color, alpha);
            }
            ENDHLSL
        }
    }
