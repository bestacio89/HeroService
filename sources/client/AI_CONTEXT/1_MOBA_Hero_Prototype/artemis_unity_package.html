Silhouette & Morphology

Tall, lean, athletic goddess — the ideal huntress form: ~180cm, long limbs built for speed and precision, not brute strength. Poised upright stance with the controlled stillness of a predator before the strike. Movement is swift and near-silent — she barely disturbs the ground when she runs. Silver-white hair flowing loose or in a high ponytail. Pale silver-blue irises that shift to radiant full-moon white during divine ability states. The bearing of an immortal who has never once missed a shot.

Divine Armor — Huntress of Olympus

Light divine armor befitting a mobile ranger: silver-white short chiton (knee-length tunic) with a gold clasped leather pauldron over the right shoulder. Silver bracers engraved with crescent moon and stag motifs. Quiver of silver-tipped arrows across the back, trailing faint moonlight as she moves. Knee-high leather hunting boots — laced, worn, practical. A crescent moon diadem in the hair, glowing subtly. The overall aesthetic is clean functional divinity: no excessive ornament, every piece earned in the hunt.

Sacred Bow — Silver Bow of Artemis

The Silver Bow — a gift from her father Zeus. A long recurve hunting bow of pure divine silver, engraved with running stags and crescent moons along the limbs. The string glows faintly with compressed moonlight when at full draw. Arrows are divine silver-tipped bodkins that trail cold white-blue light in flight, never missing a marked target. The bow is the center of her identity: she fights at range with divine precision, controlling the field through placement, angles, and the threat of the next arrow.

Sacred Companions — Deer & Hunting Hounds

Artemis is never truly alone. Two sacred silver stags accompany her as environmental companions — they circle the arena perimeter during normal play, but during the Ultimate they charge through the battlefield as divine weapons. Golden hunting hounds materialize during ability activations, biting and tracking marked targets. The stag silhouettes are visible at all times as ambient presence — glowing faintly in the tree line of the map, disappearing and reappearing with divine grace.

System Proposal

Artemis is a precision Marksman / Ranger with the longest basic attack range in the roster — befitting the goddess whose arrows never miss. She uses a Moonlight resource that charges continuously and surges on kills and marked-target hits, rewarding deliberate aggressive play.

Her kit rewards careful target selection and setup: Mark a target with Skill 1 to unlock homing arrows and guaranteed crits against them, lock them down with Skill 2's snare, then pour damage in from maximum range. The Ultimate shifts her into a full-area god-mode that makes her a teamfight deterrent — a rain of silver arrows that no formation can stand in.
Hero: 15 Levels Skill 1: 4 Ranks Skill 2: 4 Ranks Ultimate: 3 Ranks Moonlight Resource Mark System
Moonlight is a float capped at 100. It regenerates passively but gains +20 per basic attack hit and +35 when an arrow hits a Marked target. The Ultimate costs 100 Moonlight (full bar). A Marked target is tracked by a Dictionary<Health, float> expiry table in ArtemisController — exactly as Vash's Tag system, but with arrow homing and guaranteed crit instead of crit doubling.

// HeroDefinition.cs
// ScriptableObject — MOBA 15-level hero data.
// Extended for Artemis: Moonlight resource, Mark debuff, passive surge thresholds,
// dual-mode Skill 2 flag, Moonfall arrow count and stag damage.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string   heroId      = "artemis";
    public string   displayName = "Artemis";
    public HeroRole role        = HeroRole.Marksman;

    [Header("Core progression — Level 1 to 15")]
    public ScaledStat maxHp          = new(2400f, 5000f);
    public ScaledStat hpRegen         = new(7.0f,  16.0f);
    public ScaledStat moonlightRegen  = new(8.0f,  18.0f);
    public ScaledStat physicalAttack  = new(155f,  315f);
    public ScaledStat magicPower      = new(80f,   180f);
    public ScaledStat physicalDefense = new(18f,   52f);
    public ScaledStat magicDefense    = new(22f,   62f);
    public ScaledStat attackSpeed     = new(1.05f, 1.50f);
    public ScaledStat critRate        = new(0.08f, 0.22f);
    public ScaledStat critDamage      = new(1.8f,  2.5f);

    [Header("Fixed properties")]
    public float moveSpeed         = 265f;
    public float basicAttackRange  = 10.0f;
    public float attackSpeedRatio  = 1.0f;
    public float maxMoonlight      = 100f;
    public float mlOnHit           = 20f;
    public float mlOnMarkedHit     = 35f;
    public float mlOnMarkedKill    = 50f;
    public float arrowProjectileSpeed = 30f;

    [Header("Passive — Mistress of Wild Things")]
    public float surgeThreshold       = 0.80f; // 80% Moonlight
    public float surgeAttackSpeedBonus = 0.25f; // +25%
    public float surgeMoveSpeedBonus   = 15f;   // flat move speed

    [Header("Skill 2 Mode")]
    public bool skillTwoIsHound = true; // false = Bramble Snare

    [Header("Moonfall — Ultimate")]
    public int   moonfallArrowCount = 40;
    public float moonfallZoneRadius = 14f;
    public float stagChargeDamage   = 320f; // base, scaled per rank
    public float stagChargeWidth    = 2.5f;

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new("Silver Arrow",
        new[] { 0f }, new[] { 0f }, new[] { 0f }, 1.05f);

    public AbilityRankData skillOne = new("Moonlit Mark",
        new[] { 160f, 200f, 240f, 280f },
        new[] { 20f,  20f,  20f,  20f  },
        new[] { 10f,  9f,   8f,   7f   }, 0.45f);

    public AbilityRankData skillTwo = new("Sacred Hunt",
        new[] { 200f, 255f, 310f, 365f },
        new[] { 30f,  30f,  30f,  30f  },
        new[] { 13f,  12f,  11f,  10f  }, 0.55f);

    public AbilityRankData ultimate = new("Moonfall",
        new[] { 65f,  88f,  115f }, // damage per arrow
        new[] { 100f, 100f, 100f },
        new[] { 100f, 88f,  76f  }, 0.20f);

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level           = level,
            maxHp           = maxHp.Evaluate(level),
            hpRegen         = hpRegen.Evaluate(level),
            moonlightRegen  = moonlightRegen.Evaluate(level),
            physicalAttack  = physicalAttack.Evaluate(level),
            magicPower      = magicPower.Evaluate(level),
            physicalDefense = physicalDefense.Evaluate(level),
            magicDefense    = magicDefense.Evaluate(level),
            attackSpeed     = attackSpeed.Evaluate(level),
            critRate        = critRate.Evaluate(level),
            critDamage      = critDamage.Evaluate(level),
            moveSpeed       = moveSpeed,
            basicAttackRange = basicAttackRange,
            attackSpeedRatio = attackSpeedRatio,
            maxMoonlight    = maxMoonlight
        };
    }
}

public enum HeroRole { Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin }

[Serializable]
public struct ScaledStat
{
    public float level1, level15;
    public ScaledStat(float l1, float l15) { level1 = l1; level15 = l15; }
    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return level == 1 ? level1 : Mathf.Lerp(level1, level15, (level - 1f) / 14f);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] moonlightCost;
    public float[] cooldown;
    public float   scalingPhysical;

    public AbilityRankData(string n, float[] dmg, float[] cost, float[] cd, float scale)
    { abilityName = n; baseDamage = dmg; moonlightCost = cost; cooldown = cd; scalingPhysical = scale; }

    public float DamageAtRank    (int r, float atq) => baseDamage  [Mathf.Clamp(r-1,0,baseDamage.Length-1)]    + atq * scalingPhysical;
    public float MlCostAtRank    (int r)          => moonlightCost[Mathf.Clamp(r-1,0,moonlightCost.Length-1)];
    public float CooldownAtRank  (int r)          => cooldown     [Mathf.Clamp(r-1,0,cooldown.Length-1)];
}

[Serializable]
public struct RuntimeHeroStats
{
    public int   level;
    public float maxHp, hpRegen;
    public float maxMoonlight, moonlightRegen;
    public float physicalAttack, magicPower, physicalDefense, magicDefense;
    public float attackSpeed, moveSpeed, basicAttackRange, attackSpeedRatio;
    public float critRate, critDamage;
}
C# ArtemisController.cs
// ArtemisController.cs
// Main controller: Moonlight resource, Marked target dictionary, homing arrow physics,
// passive surge state (80%+ ML), dual-mode Skill 2 (hound vs bramble),
// Moonfall 40-arrow staggered volley + stag charges, untargetable channel.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;
using System.Collections.Generic;

[RequireComponent(typeof(NavMeshAgent))]
public class ArtemisController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator          animator;
    [SerializeField] private NavMeshAgent      agent;
    [SerializeField] private ArtemisCombatVFX  combatVfx;
    [SerializeField] private Transform          bowTip;
    [SerializeField] private Transform          leftStagSpawn;
    [SerializeField] private Transform          rightStagSpawn;
    [SerializeField] private LayerMask          enemyMask;
    [SerializeField] private LayerMask          wallMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentMoonlight;
    [SerializeField] private bool  isDead;
    [SerializeField] private bool  isUntargetable;
    [SerializeField] private bool  surgeActive;

    private float basicCooldown, skillOneCooldown, skillTwoCooldown, ultimateCooldown;
    private readonly Dictionary<Health, float> markedTargets = new(); // target → expiry time

    public float HpPercent        => stats.maxHp        > 0f ? currentHp        / stats.maxHp        : 0f;
    public float MoonlightPercent  => stats.maxMoonlight > 0f ? currentMoonlight / stats.maxMoonlight : 0f;
    public bool  SurgeActive       => surgeActive;
    public bool  IsUntargetable    => isUntargetable;
    public bool  IsDead            => isDead;

    private void Reset() { agent = GetComponent<NavMeshAgent>(); animator = GetComponentInChildren<Animator>(); }
    private void Awake() { if (agent == null) agent = GetComponent<NavMeshAgent>(); ApplyLevel(heroLevel, true); }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        RegenerateResources();
        CheckSurgePassive();
        PruneExpiredMarks();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore) { currentHp = stats.maxHp; currentMoonlight = 0f; }
        else currentHp = Mathf.Min(currentHp, stats.maxHp);
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
    }

    // ── BASIC ATTACK — silver arrow with homing on Marked target ──────────
    public void BasicAttack(Health target)
    {
        if (isDead || basicCooldown > 0f || target == null) return;

        float atkSpeed  = GetEffectiveAttackSpeed();
        basicCooldown   = 1f / Mathf.Max(0.1f, atkSpeed * stats.attackSpeedRatio);
        bool  isMarked  = IsMarked(target);
        bool  forceCrit = isMarked; // guaranteed crit on Marked target
        float dmg       = ResolveDamage(heroDefinition.basicAttack.DamageAtRank(1, stats.physicalAttack), forceCrit);

        StartCoroutine(ArrowProjectileRoutine(target, dmg, isMarked));
        animator.SetTrigger("BasicAttack");
    }

    private IEnumerator ArrowProjectileRoutine(Health target, float dmg, bool marked)
    {
        Vector3 startPos = bowTip.position;
        Vector3 targetPos = target.transform.position + Vector3.up * 1.0f;
        float   dist      = Vector3.Distance(startPos, targetPos);
        float   travelTime = dist / heroDefinition.arrowProjectileSpeed;

        combatVfx.PlayArrowFire(startPos, targetPos, marked);
        yield return new WaitForSeconds(travelTime);

        if (target == null) yield break;
        target.ApplyPhysicalDamage(dmg);
        if (marked && dmg > stats.physicalAttack) combatVfx.PlayCritFlash(target.transform.position);

        // Moonlight gain
        GainMoonlight(marked ? heroDefinition.mlOnMarkedHit : heroDefinition.mlOnHit);
    }

    // ── SKILL 1 — Moonlit Mark ────────────────────────────────────────────
    public bool TryCastSkillOne(Health target)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown) || target == null) return false;
        SpendMoonlight(heroDefinition.skillOne.MlCostAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        animator.SetTrigger("CastMark");
        StartCoroutine(MarkArrowRoutine(target, skillOneRank));
        return true;
    }

    private IEnumerator MarkArrowRoutine(Health target, int rank)
    {
        Vector3 targetPos = target.transform.position + Vector3.up;
        float travelTime  = Vector3.Distance(bowTip.position, targetPos) / heroDefinition.arrowProjectileSpeed;
        combatVfx.PlayMarkArrow(bowTip.position, targetPos);
        yield return new WaitForSeconds(travelTime);
        if (target == null) yield break;

        float dmg         = heroDefinition.skillOne.DamageAtRank(rank, stats.physicalAttack);
        float markDuration = 4.5f + rank * 0.5f;
        float slowDuration = 0.8f + rank * 0.2f;

        target.ApplyPhysicalDamage(dmg);
        target.GetComponent<CrowdControlReceiver>()?.ApplySlow(0.30f, slowDuration);

        // Transfer mark (remove all existing, apply to new target)
        markedTargets.Clear();
        markedTargets[target] = Time.time + markDuration;
        combatVfx.PlayMarkSigil(target.transform.position, markDuration);
        GainMoonlight(heroDefinition.mlOnHit);
    }

    // ── SKILL 2 — Dual Mode: Hunting Hound OR Bramble Snare ──────────────
    public bool TryCastSkillTwo(Health target, Vector3 groundTarget)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;
        SpendMoonlight(heroDefinition.skillTwo.MlCostAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);
        animator.SetTrigger(heroDefinition.skillTwoIsHound ? "SummonHound" : "BrambleSnare");

        if (heroDefinition.skillTwoIsHound && target != null)
            StartCoroutine(HuntingHoundRoutine(target, skillTwoRank));
        else
            StartCoroutine(BrambleSnareRoutine(groundTarget, skillTwoRank));
        return true;
    }

    private IEnumerator HuntingHoundRoutine(Health target, int rank)
    {
        combatVfx.PlayHoundSummon(transform.position, target.transform.position);
        float houndSpeed = 14f;
        float dist       = Vector3.Distance(transform.position, target.transform.position);
        yield return new WaitForSeconds(dist / houndSpeed);
        if (target == null) yield break;
        float dmg          = heroDefinition.skillTwo.DamageAtRank(rank, stats.physicalAttack);
        float slowDuration = 1.8f + rank * 0.2f;
        target.ApplyPhysicalDamage(dmg);
        target.GetComponent<CrowdControlReceiver>()?.ApplySlow(0.40f, slowDuration);
        combatVfx.PlayHoundBite(target.transform.position);
    }

    private IEnumerator BrambleSnareRoutine(Vector3 center, int rank)
    {
        yield return new WaitForSeconds(0.25f);
        float radius      = 3.5f;
        float rootDuration = 1.3f + rank * 0.3f;
        float dmg         = heroDefinition.skillTwo.DamageAtRank(rank, stats.physicalAttack) * 0.6f; // bramble mode lower dmg
        combatVfx.PlayBrambleSnare(center, radius, rootDuration);
        foreach (var col in Physics.OverlapSphere(center, radius, enemyMask))
        {
            col.GetComponent<Health>()?.ApplyPhysicalDamage(dmg);
            col.GetComponent<CrowdControlReceiver>()?.ApplyRoot(rootDuration);
        }
    }

    // ── ULTIMATE — Moonfall: 40-arrow rain + stag charge ──────────────────
    public bool TryCastUltimate(Vector3 zoneCenter)
    {
        if (isDead || ultimateCooldown > 0f) return false;
        if (currentMoonlight < stats.maxMoonlight) return false; // must be full

        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        currentMoonlight = 0f;
        StartCoroutine(MoonfallRoutine(zoneCenter, ultimateRank));
        return true;
    }

    private IEnumerator MoonfallRoutine(Vector3 center, int rank)
    {
        // Untargetable channel — step into moonlight
        isUntargetable  = true;
        agent.isStopped = true;
        animator.SetTrigger("MoonfallChannel");
        combatVfx.StartMoonfallChannel(transform.position);

        float channelTime = rank >= 3 ? 1.2f : 1.5f;
        yield return new WaitForSeconds(channelTime);

        isUntargetable = false;
        animator.SetTrigger("MoonfallFire");

        float   zone         = heroDefinition.moonfallZoneRadius + rank;
        float   arrowDmg     = heroDefinition.ultimate.DamageAtRank(rank, stats.physicalAttack);
        float   stagDmg      = heroDefinition.stagChargeDamage + rank * 100f;
        int     arrowCount   = heroDefinition.moonfallArrowCount;
        float   volleyDur    = rank >= 3 ? 1.6f : 2.0f;
        float   interval     = volleyDur / arrowCount;

        // Find any marked target in zone for homing arrows
        Health markedInZone = FindMarkedInZone(center, zone);
        combatVfx.PlayMoonfallRain(center, zone, arrowCount, volleyDur);

        // Stagger individual arrow damage over volley duration
        for (int i = 0; i < arrowCount; i++)
        {
            yield return new WaitForSeconds(interval);
            // Each arrow hits a random point in the zone
            Vector3 randPoint = center + new Vector3(
                Random.Range(-zone, zone), 0,
                Random.Range(-zone, zone));

            // Homing bias toward marked target
            if (markedInZone != null && i % 3 == 0)
                randPoint = markedInZone.transform.position;

            foreach (var col in Physics.OverlapSphere(randPoint, 1.8f, enemyMask))
                col.GetComponent<Health>()?.ApplyPhysicalDamage(ResolveDamage(arrowDmg, true)); // all arrows crit
        }

        // Sacred stag charges — two lines through zone from opposite sides
        StartCoroutine(StagChargeRoutine(center, zone, stagDmg, true));
        StartCoroutine(StagChargeRoutine(center, zone, stagDmg, false));

        yield return new WaitForSeconds(volleyDur + 0.5f);
        combatVfx.StopMoonfallChannel();
        agent.isStopped = false;
    }

    private IEnumerator StagChargeRoutine(Vector3 center, float radius, float dmg, bool leftToRight)
    {
        Vector3 startPos = center + (leftToRight ? Vector3.left  : Vector3.right) * (radius + 3f);
        Vector3 endPos   = center + (leftToRight ? Vector3.right : Vector3.left)  * (radius + 3f);
        Vector3 dir      = (endPos - startPos).normalized;
        float   chargeSpeed = 20f;
        float   duration    = (radius * 2f + 6f) / chargeSpeed;

        combatVfx.PlayStagCharge(startPos, dir, duration);

        float elapsed = 0f;
        Vector3 pos   = startPos;
        var hit = new HashSet<Health>();
        while (elapsed < duration)
        {
            pos += dir * chargeSpeed * Time.deltaTime;
            foreach (var col in Physics.OverlapSphere(pos, heroDefinition.stagChargeWidth, enemyMask))
            {
                var hp = col.GetComponent<Health>();
                if (hp != null && hit.Add(hp))
                {
                    hp.ApplyPhysicalDamage(dmg);
                    col.GetComponent<CrowdControlReceiver>()?.ApplyKnockback(dir, 2.5f);
                }
            }
            elapsed += Time.deltaTime;
            yield return null;
        }
    }

    // ── DAMAGE RECEPTION — untargetable during channel ─────────────────────
    public void ReceiveDamage(float amount)
    {
        if (isDead || isUntargetable) return; // divine invulnerability during Moonfall channel
        currentHp = Mathf.Max(0f, currentHp - amount);
        animator.SetTrigger("Hit");

        if (currentHp <= 0f) StartCoroutine(DeathRoutine());

    private IEnumerator DeathRoutine()
    {
        isDead = true;
        agent.isStopped = true;
        animator.SetBool("IsDead", true);
        combatVfx.PlayDeath(transform.position);
        yield return new WaitForSeconds(3.5f);

    // ── PASSIVE — Moonlight Surge ──────────────────────────────────────────
    private void CheckSurgePassive()
    {
        bool shouldSurge = MoonlightPercent >= heroDefinition.surgeThreshold;
        if (shouldSurge == surgeActive) return;
        surgeActive = shouldSurge;
        float speedMult = surgeActive ? (1f + heroDefinition.surgeAttackSpeedBonus) : 1f;
        float moveAdd   = surgeActive ? heroDefinition.surgeMoveSpeedBonus : 0f;
        agent.speed = (stats.moveSpeed + moveAdd) / 100f;
        combatVfx.SetSurgeGlow(surgeActive);
    }

    private float GetEffectiveAttackSpeed()
        => surgeActive ? stats.attackSpeed * (1f + heroDefinition.surgeAttackSpeedBonus) : stats.attackSpeed;

    private float ResolveDamage(float base_, bool forceCrit)
    {
        bool  isCrit    = forceCrit || Random.value < stats.critRate;
        float mult      = isCrit ? stats.critDamage : 1f;
        if   (isCrit) combatVfx.PlayCritFlash(Vector3.zero);
        return base_ * mult;
    }

    public bool    IsMarked(Health h)   => h != null && markedTargets.TryGetValue(h, out float t) && Time.time < t;
    private Health FindMarkedInZone(Vector3 c, float r)
    {
        foreach (var col in Physics.OverlapSphere(c, r, enemyMask))
        { var h = col.GetComponent<Health>(); if (IsMarked(h)) return h; }
        return null;
    }

    private void PruneExpiredMarks()
    {
        var expired = new List<Health>();
        foreach (var kv in markedTargets) if (Time.time >= kv.Value) expired.Add(kv.Key);
        foreach (var k  in expired)       markedTargets.Remove(k);
    }

    private void GainMoonlight(float a) => currentMoonlight = Mathf.Min(stats.maxMoonlight, currentMoonlight + a);
    private void SpendMoonlight(float a) => currentMoonlight = Mathf.Max(0f, currentMoonlight - a);
    private bool  CanCast(AbilityRankData ab, int r, float cd)
        => !isDead && cd <= 0f && currentMoonlight >= ab.MlCostAtRank(r);

    private void TickCooldowns()
    {
        basicCooldown    = Mathf.Max(0f, basicCooldown    - Time.deltaTime);
        skillOneCooldown = Mathf.Max(0f, skillOneCooldown - Time.deltaTime);
        skillTwoCooldown = Mathf.Max(0f, skillTwoCooldown - Time.deltaTime);
        ultimateCooldown = Mathf.Max(0f, ultimateCooldown - Time.deltaTime);
    }

    private void RegenerateResources()
    {
        currentHp         = Mathf.Min(stats.maxHp, currentHp + stats.hpRegen * Time.deltaTime);
        currentMoonlight  = Mathf.Min(stats.maxMoonlight, currentMoonlight + stats.moonlightRegen * Time.deltaTime);
        combatVfx.UpdateHeroState(HpPercent, MoonlightPercent, surgeActive, agent.velocity.sqrMagnitude > 0.01f);
    }

    private void UpdateAnimator()
    {
        Vector3 vel = agent.velocity; vel.y = 0f;
        animator.SetFloat("MoveSpeed",       vel.magnitude);
        animator.SetFloat("AttackSpeed",     GetEffectiveAttackSpeed());
        animator.SetFloat("MoonlightPercent", MoonlightPercent);
        animator.SetBool("SurgeActive",      surgeActive);
        animator.SetBool("Untargetable",     isUntargetable);
    }

                float  alpha  = sigil.a * _MarkPercent * pulse * _SigilColor.a;

                return half4(color * sigil.rgb, alpha);
            }
            ENDHLSL
        }
    }
}
HLSL MoonfallGroundShimmer.shader
// MoonfallGroundShimmer.shader
// Full-circle ground effect during Moonfall — unlike Vash's semicircle,
// Artemis's zone is a complete circle. Rolling silver-white shimmer spreads
// inward from the radius edge as arrows rain down, fading as the volley ends.
// _FillPercent driven by CPU over the volley duration (0=start, 1=fully active).

Shader "Artemis/MoonfallGroundShimmer"
{
    Properties
    {
        _MoonColor  ("Moon Silver Color", Color)       = (0.80, 0.92, 1.00, 0.50)
        _EdgeColor  ("Edge Glow Color",   Color)       = (1.00, 1.00, 1.00, 0.90)
        _Radius     ("Zone Radius",       Float)       = 14.0
        _FillPercent("Fill %",            Range(0,1))  = 0.0
        _RippleSpeed("Ripple Speed",      Range(0,6))  = 2.8
        _RippleCount("Ripple Rings",      Range(1,8))  = 4.0
        _FadeOut    ("Fade Out",          Range(0,1))  = 0.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _MoonColor;
                float4 _EdgeColor;
                float  _Radius;
                float  _FillPercent;
                float  _RippleSpeed;
                float  _RippleCount;
                float  _FadeOut;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                OUT.positionCS = GetVertexPositionInputs(IN.positionOS.xyz).positionCS;
                OUT.uv = IN.uv * 2.0 - 1.0;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float dist = length(IN.uv); // 0=center, 1=edge at radius

                // Zone fill boundary — expands inward from edge as FillPercent rises
                float boundary = 1.0 - _FillPercent;
                float fill     = smoothstep(boundary + 0.05, boundary, dist);

                // Concentric ripple rings spreading inward (arrows hitting ground)
                float ripple   = 0.5 + 0.5 * sin((dist * _RippleCount - _Time.y * _RippleSpeed) * 6.2831);
                ripple         = smoothstep(0.3, 0.7, ripple) * fill;

                // Bright edge ring at zone boundary
                float edgeRing = smoothstep(1.0, 0.88, dist) * smoothstep(0.80, 1.0, dist);

                float fade      = saturate(1.0 - _FadeOut);
                half3 color     = lerp(_MoonColor.rgb, _EdgeColor.rgb, ripple * 0.4 + edgeRing * 0.6);
                float alpha     = (fill * _MoonColor.a * 0.4 + ripple * 0.35 + edgeRing * _EdgeColor.a) * fade;

                return half4(color, saturate(alpha));
            }
            ENDHLSL
        }
    }
}
