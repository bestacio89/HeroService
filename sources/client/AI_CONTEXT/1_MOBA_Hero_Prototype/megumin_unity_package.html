Silhouette & Morphology

Small, slim young arch wizard — short stature (152cm) that wildly contradicts her raw destructive power. Slightly hunched theatrical posture when casting — she leans dramatically into every incantation. Moves with energetic, exaggerated anime flair: wide gestures, sweeping staff arcs, the iconic post-explosion collapse to the ground with zero MP remaining. Her tiny frame against screen-filling explosion VFX is the entire aesthetic.

Costume — Crimson Demon Arch Wizard

Distinctive dark witch ensemble: black pointed hat with a tattered wide brim and a bright red ribbon band. Black cape over the shoulders with a deep crimson inner lining. Short black dress with a white collar undershirt and red ribbon bow at the neck. Black thigh-high stockings, black boots. Red gem eye-patch over the right eye — worn purely for aesthetic effect, nothing wrong with the eye. Short messy chestnut-brown hair. Crimson red irises. Staff always in hand or at hip.

Staff — Arch Wizard's Rod

Elegant dark wood staff, roughly Megumin's height. The top terminates in a curved ornamental head inset with a large deep-crimson gemstone that pulses with magical energy during charge-up. The gem brightens dramatically as Explosion magic builds — near-blinding at full charge. Purple-red magic circles orbit the staff tip during incantation. The staff is her entire identity; she would sooner collapse than set it down.

The One-Cast Rule — Core Design Constraint

The single most defining mechanical trait in all of KonoSuba: Megumin can cast Explosion exactly once per battle before her MP is completely depleted. Post-cast she is entirely helpless — cannot walk, falls over, must be carried. In MOBA terms this becomes a unique MP floor mechanic: she can use minor spells to build toward Explosion, but the moment Explosion fires, her MP drops to zero and she enters a Downed State lasting several seconds — the highest single-ability damage in the roster at the cost of total vulnerability.

System Proposal

Megumin is a glass-cannon burst mage with an entirely unique resource design: her MP pool is fixed at 300 at all levels — canonically because she put every single stat point into Explosion. MP regenerates over time but casting the Ultimate costs all 300 MP at once, leaving her at zero and triggering the Downed State.

Minor skills (1 and 2) cost small MP amounts to keep Explosion within reach. The trade-off is extreme: when Explosion fires it deals the highest single-hit damage of any ability in the roster at the cost of complete post-cast helplessness. High risk, maximum reward — exactly as the source material demands.

C# HeroDefinition.cs
// HeroDefinition.cs
// ScriptableObject — MOBA 15-level hero data.
// Extended for Megumin: fixed MP pool, Explosion passive multipliers,
// Downed State duration, full-bar bonus thresholds.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string   heroId      = "megumin";
    public string   displayName = "Megumin";
    public HeroRole role        = HeroRole.Mage;

    [Header("Core progression — Level 1 to 15")]
    public ScaledStat maxHp          = new(1800f, 3800f);
    public ScaledStat hpRegen         = new(4.5f,  10.0f);
    public ScaledStat mpRegen          = new(18.0f, 38.0f);
    public ScaledStat magicPower       = new(220f,  560f);
    public ScaledStat physicalDefense  = new(10f,   32f);
    public ScaledStat magicDefense     = new(28f,   72f);
    public ScaledStat attackSpeed      = new(0.60f, 0.80f);

    [Header("Fixed properties")]
    public float moveSpeed          = 235f;
    public float basicAttackRange   = 1.8f;
    public float attackSpeedRatio   = 1.0f;
    public float maxMp              = 300f; // FIXED — all points in Explosion
    public float explosionRange     = 22f;
    public float tapMpBonus         = 15f; // MP gained after 3 staff taps
    public float missileMpFeedback  = 5f;  // MP on Magic Missile hit

    [Header("Passive — All Points in Explosion")]
    public float explosionApMultiplier  = 2.0f; // double AP scaling on Ultimate
    public float fullBarDamageBonus     = 0.10f; // +10% damage when cast at full MP
    public float fullBarRadiusBonus     = 0.15f; // +15% radius when cast at full MP
    public float channelReductionPerRank = 0.3f; // seconds removed per ultimate rank

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new("Staff Tap",
        new[] { 0f }, new[] { 0f }, new[] { 0f }, 0.30f);

    public AbilityRankData skillOne = new("Magic Missile",
        new[] { 160f, 210f, 260f, 310f },
        new[] { 20f,  20f,  20f,  20f  },
        new[] { 5f,   4.5f, 4f,   3.5f }, 0.55f);

    public AbilityRankData skillTwo = new("Dark Magic Bind",
        new[] { 90f,  120f, 150f, 180f },
        new[] { 40f,  40f,  40f,  40f  },
        new[] { 14f,  13f,  12f,  11f  }, 0.40f);

    public AbilityRankData ultimate = new("Explosion",
        new[] { 1800f, 2400f, 3200f },
        new[] { 300f,  300f,  300f  }, // always costs full bar
        new[] { 90f,   80f,   70f   }, 0f); // scaling handled by passive ×2.0 AP override

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level           = level,
            maxHp           = maxHp.Evaluate(level),
            hpRegen         = hpRegen.Evaluate(level),
            mpRegen         = mpRegen.Evaluate(level),
            magicPower      = magicPower.Evaluate(level),
            physicalDefense = physicalDefense.Evaluate(level),
            magicDefense    = magicDefense.Evaluate(level),
            attackSpeed     = attackSpeed.Evaluate(level),
            moveSpeed       = moveSpeed,
            basicAttackRange = basicAttackRange,
            attackSpeedRatio = attackSpeedRatio,
            maxMp           = maxMp  // always 300
        };
    }
}

public enum HeroRole { Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin }

[Serializable]
public struct ScaledStat
{
    public float level1, level15;
    public ScaledStat(float l1, float l15) { level1 = l1; level15 = l15; }
    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return level == 1 ? level1 : Mathf.Lerp(level1, level15, (level - 1f) / 14f);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] mpCost;
    public float[] cooldown;
    public float   scalingMagicPower;

    public AbilityRankData(string n, float[] dmg, float[] cost, float[] cd, float scale)
    { abilityName = n; baseDamage = dmg; mpCost = cost; cooldown = cd; scalingMagicPower = scale; }

    public float DamageAtRank (int r, float ap) => baseDamage[Mathf.Clamp(r-1,0,baseDamage.Length-1)] + ap * scalingMagicPower;
    public float MpCostAtRank (int r)         => mpCost    [Mathf.Clamp(r-1,0,mpCost.Length-1)];
    public float CooldownAtRank(int r)         => cooldown  [Mathf.Clamp(r-1,0,cooldown.Length-1)];
}

[Serializable]
public struct RuntimeHeroStats
{
    public int   level;
    public float maxHp, hpRegen;
    public float maxMp, mpRegen;
    public float magicPower, physicalDefense, magicDefense;
    public float attackSpeed, moveSpeed, basicAttackRange, attackSpeedRatio;
}
C# MeguminController.cs
// MeguminController.cs
// Main controller: fixed 300-MP pool, Explosion incantation channel with interrupt,
// post-Explosion Downed State, ally proximity early-cancel, full-bar bonus check,
// staff-tap combo MP refund, Magic Missile feedback MP, Dark Magic Bind root.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class MeguminController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator          animator;
    [SerializeField] private NavMeshAgent      agent;
    [SerializeField] private MeguminCombatVFX  combatVfx;
    [SerializeField] private Transform          staffTip;
    [SerializeField] private LayerMask          enemyMask;
    [SerializeField] private LayerMask          allyMask;

    [Header("Downed State")]
    [SerializeField] private float downedDuration       = 4.0f; // reduced at higher ult ranks
    [SerializeField] private float allyRescueRange       = 2.0f; // ally proximity cancel range

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentMp;
    [SerializeField] private bool  isDead;
    [SerializeField] private bool  isDowned;
    [SerializeField] private bool  isChanneling;

    private float basicCooldown, skillOneCooldown, skillTwoCooldown, ultimateCooldown;
    private int   tapComboCount = 0;
    private float tapComboTimer = 0f;
    private const float TAP_COMBO_WINDOW = 2.0f;
    private Coroutine channelingRoutine;

    public float HpPercent  => stats.maxHp > 0f ? currentHp / stats.maxHp : 0f;
    public float MpPercent  => stats.maxMp > 0f ? currentMp / stats.maxMp : 0f;
    public bool  IsFullMp   => currentMp >= stats.maxMp - 0.5f;
    public bool  IsDowned   => isDowned;
    public bool  IsChanneling => isChanneling;
    public bool  IsDead     => isDead;

    private void Reset() { agent = GetComponent<NavMeshAgent>(); animator = GetComponentInChildren<Animator>(); }
    private void Awake() { if (agent == null) agent = GetComponent<NavMeshAgent>(); ApplyLevel(heroLevel, true); }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        RegenerateResources();
        TickTapCombo();
        if (isDowned) CheckAllyRescue();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore) { currentHp = stats.maxHp; currentMp = 0f; } // starts with 0 MP — must regen
        else currentHp = Mathf.Min(currentHp, stats.maxHp);
        downedDuration = 4.5f - ultimateRank * 0.5f; // 4.0 / 3.5 / 3.0
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead || isDowned || isChanneling) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
    }

    // ── BASIC ATTACK — staff tap (comedic fallback) ───────────────────────
    public void BasicAttack(Health target)
    {
        if (isDead || isDowned || basicCooldown > 0f || target == null) return;
        basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
        float dmg = stats.magicPower * heroDefinition.basicAttack.scalingMagicPower;
        target.ApplyMagicDamage(dmg);
        animator.SetTrigger("BasicAttack");
        combatVfx.PlayStaffTap(staffTip.position);

        // 3-tap combo MP bonus
        tapComboTimer = TAP_COMBO_WINDOW;
        tapComboCount++;
        if (tapComboCount >= 3)
        {
            tapComboCount = 0;
            GainMp(heroDefinition.tapMpBonus);
            combatVfx.PlayMpFeedback(transform.position);
        }
    }

    // ── SKILL 1 — Magic Missile ───────────────────────────────────────────
    public bool TryCastSkillOne(Vector3 targetPosition)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;
        SpendMp(heroDefinition.skillOne.MpCostAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        animator.SetTrigger("CastSkill01");
        StartCoroutine(MagicMissileRoutine(targetPosition, skillOneRank));
        return true;
    }

    private IEnumerator MagicMissileRoutine(Vector3 targetPos, int rank)
    {
        Vector3 dir = (targetPos - staffTip.position).normalized;
        combatVfx.PlayMagicMissile(staffTip.position, dir);
        float travelTime = Vector3.Distance(staffTip.position, targetPos) / 20f;
        yield return new WaitForSeconds(travelTime);

        foreach (var col in Physics.OverlapSphere(targetPos, 0.8f, enemyMask))
        {
            var hp = col.GetComponent<Health>();
            if (hp != null)
            {
                float dmg = heroDefinition.skillOne.DamageAtRank(rank, stats.magicPower);
                hp.ApplyMagicDamage(dmg);
                // Arcane Feedback — gain MP on hit
                GainMp(heroDefinition.missileMpFeedback);
                combatVfx.PlayMpFeedback(transform.position);
            }
        }
        combatVfx.PlayMissileImpact(targetPos);
    }

    // ── SKILL 2 — Dark Magic Bind ─────────────────────────────────────────
    public bool TryCastSkillTwo(Vector3 groundTarget)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;
        SpendMp(heroDefinition.skillTwo.MpCostAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);
        animator.SetTrigger("CastSkill02");
        StartCoroutine(DarkBindRoutine(groundTarget, skillTwoRank));
        return true;
    }

    private IEnumerator DarkBindRoutine(Vector3 center, int rank)
    {
        yield return new WaitForSeconds(0.2f);
        float radius      = 4.5f;
        float rootDuration = 1.2f + rank * 0.3f;
        float dmg         = heroDefinition.skillTwo.DamageAtRank(rank, stats.magicPower);
        combatVfx.PlayDarkBind(center, radius, rootDuration);
        foreach (var col in Physics.OverlapSphere(center, radius, enemyMask))
        {
            col.GetComponent<Health>()?.ApplyMagicDamage(dmg);
            col.GetComponent<CrowdControlReceiver>()?.ApplyRoot(rootDuration);
        }
    }

    // ── ULTIMATE — EXPLOSION: incantation channel + blast + Downed State ──
    public bool TryCastUltimate(Vector3 blastTarget)
    {
        if (isDead || isDowned || isChanneling || ultimateCooldown > 0f) return false;
        if (currentMp < stats.maxMp) return false; // must have full 300 MP

        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        channelingRoutine = StartCoroutine(ExplosionRoutine(blastTarget, ultimateRank));
        return true;
    }

    // Cancel incantation (e.g. interrupted by enemy CC) — returns 50% MP
    public void InterruptChannel()
    {
        if (!isChanneling || channelingRoutine == null) return;
        StopCoroutine(channelingRoutine);
        isChanneling = false;
        agent.isStopped = false;
        GainMp(stats.maxMp * 0.5f); // 50% refund on interrupt
        animator.SetTrigger("Interrupt");
        combatVfx.StopExplosionChannel();
    }

    private IEnumerator ExplosionRoutine(Vector3 blastCenter, int rank)
    {
        isChanneling = true;
        agent.isStopped = true;
        animator.SetTrigger("BeginIncantation");
        combatVfx.StartExplosionChannel(transform.position);

        // Incantation channel — duration reduced by passive per rank
        float channelTime = 2.5f - (rank - 1) * heroDefinition.channelReductionPerRank;
        yield return new WaitForSeconds(channelTime);

        isChanneling = false;

        // Passive full-bar check
        bool  fullBar     = IsFullMp;
        float blastRadius = 11f + rank;
        if (fullBar) blastRadius *= (1f + heroDefinition.fullBarRadiusBonus);

        // Base damage uses ×2.0 AP multiplier from passive
        float baseDmg = heroDefinition.ultimate.DamageAtRank(rank, 0f); // base only
        float apDmg   = stats.magicPower * heroDefinition.explosionApMultiplier;
        float totalDmg = baseDmg + apDmg;
        if (fullBar) totalDmg *= (1f + heroDefinition.fullBarDamageBonus);

        // Drain ALL MP before firing
        currentMp = 0f;

        // EXPLOSION fires
        animator.SetTrigger("FireExplosion");
        combatVfx.PlayExplosion(blastCenter, blastRadius, rank);

        foreach (var col in Physics.OverlapSphere(blastCenter, blastRadius, enemyMask))
            col.GetComponent<Health>()?.ApplyMagicDamage(totalDmg); // no fall-off

        // Brief detonation pause then Downed State
        yield return new WaitForSeconds(0.3f);
        StartCoroutine(DownedRoutine());
    }

    // ── DOWNED STATE — post-explosion collapse ────────────────────────────
    private IEnumerator DownedRoutine()
    {
        isDowned = true;
        agent.isStopped = true;
        animator.SetBool("IsDowned", true);
        combatVfx.PlayDownedState(transform.position);

        float elapsed = 0f;
        while (elapsed < downedDuration)
        {
            if (CheckAllyRescue()) yield break; // ally rescued early
            elapsed += Time.deltaTime;
            yield return null;
        }
        EndDownedState();
    }

    private bool CheckAllyRescue()
    {
