Proposition de système

Kenshin est un Assassin / Fighter hybride à 15 niveaux avec croissance agressive de l'Attaque physique compensée par une défense modérée. Son passif Mode Battōsai amplifie ses dégâts sans coût mana. La mana suit le standard 500 → 1900 avec régénération légèrement supérieure pour soutenir son style agressif courte portée.

Les dégâts de compétences scalent entièrement sur l'Attaque physique (pas de magic power), ce qui rend les items offensifs physiques directement efficaces sur toutes ses capacités.

// HeroDefinition.cs
// ScriptableObject data-driven — modèle MOBA 15 niveaux
// Toutes les stats et rangs de compétences de Kenshin Himura.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string heroId      = "kenshin";
    public string displayName = "Kenshin Himura";
    public HeroRole role      = HeroRole.AssassinFighter;

    [Header("Core progression — Niveau 1 à 15")]
    public ScaledStat maxHp          = new(2850f,  6100f);
    public ScaledStat hpRegen        = new(8.5f,   18.3f);
    public ScaledStat maxMana        = new(500f,   1900f);
    public ScaledStat manaRegen      = new(4.5f,    7.8f);
    public ScaledStat physicalAttack = new(138f,    298f);
    public ScaledStat magicPower     = new(0f,        0f);
    public ScaledStat physicalDefense= new(22f,      68f);
    public ScaledStat magicDefense   = new(12f,      46f);
    public ScaledStat attackSpeed    = new(1.15f,   1.60f);

    [Header("Fixed hero properties")]
    public float moveSpeed         = 275f;
    public float basicAttackRange  = 2.8f;
    public float attackSpeedRatio  = 1.0f;
    public float criticalDamage    = 2.15f;

    [Header("Passif — Battōsai Mode (scaling par niveau)")]
    public ScaledStat battosaiAtkBonus = new(0.20f, 0.35f);  // multiplicateur ATQ
    public ScaledStat battosaiAtkSpeed = new(0.20f, 0.32f);  // bonus flat AS
    public ScaledStat battosaiBleeding = new(14f,   30f);    // DoT / s

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new(
        "Frappe du Sakabato",
        new[] { 0f },
        new[] { 0f },
        new[] { 0f },
        0f);

    public AbilityRankData skillOne = new(
        "Ryusosen",
        new[] { 48f, 62f, 76f, 90f },
        new[] { 60f, 65f, 70f, 75f },
        new[] {  9f,  8f,  7f,  6f },
        0.55f);

    public AbilityRankData skillTwo = new(
        "Doryusen",
        new[] { 180f, 240f, 300f, 360f },
        new[] {  65f,  75f,  85f,  95f },
        new[] {  13f,  12f,  11f,  10f },
        0.65f);

    public AbilityRankData ultimate = new(
        "Amakakeru Ryu no Hirameki",
        new[] {  650f,  900f, 1200f },
        new[] {  130f,  160f,  190f },
        new[] {  100f,   90f,   80f },
        1.80f);

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level            = level,
            maxHp            = maxHp.Evaluate(level),
            hpRegen          = hpRegen.Evaluate(level),
            maxMana          = maxMana.Evaluate(level),
            manaRegen        = manaRegen.Evaluate(level),
            physicalAttack   = physicalAttack.Evaluate(level),
            magicPower       = magicPower.Evaluate(level),
            physicalDefense  = physicalDefense.Evaluate(level),
            magicDefense     = magicDefense.Evaluate(level),
            attackSpeed      = attackSpeed.Evaluate(level),
            moveSpeed        = moveSpeed,
            basicAttackRange = basicAttackRange,
            attackSpeedRatio = attackSpeedRatio,
            criticalDamage   = criticalDamage
        };
    }
}

public enum HeroRole
{
    Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin, AssassinFighter
}

[Serializable]
public struct ScaledStat
{
    public float level1;
    public float level15;

    public ScaledStat(float level1, float level15)
    {
        this.level1  = level1;
        this.level15 = level15;
    }

    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        if (level == 1) return level1;
        float t = (level - 1f) / 14f;
        return Mathf.Lerp(level1, level15, t);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] manaCost;
    public float[] cooldown;
    public float   scalingPhysicalAttack;

    public AbilityRankData(string abilityName, float[] baseDamage,
                           float[] manaCost, float[] cooldown, float scalingPhysicalAttack)
    {
        this.abilityName          = abilityName;
        this.baseDamage           = baseDamage;
        this.manaCost             = manaCost;
        this.cooldown             = cooldown;
        this.scalingPhysicalAttack= scalingPhysicalAttack;
    }

    public float DamageAtRank(int rank, float physicalAttack)
        => baseDamage[Mathf.Clamp(rank - 1, 0, baseDamage.Length - 1)]
         + physicalAttack * scalingPhysicalAttack;

    public float ManaAtRank(int rank)
        => manaCost[Mathf.Clamp(rank - 1, 0, manaCost.Length - 1)];

    public float CooldownAtRank(int rank)
        => cooldown[Mathf.Clamp(rank - 1, 0, cooldown.Length - 1)];
}

[Serializable]

    public float physicalDefense;
    public float magicDefense;
    public float attackSpeed;
    public float moveSpeed;
    public float basicAttackRange;
    public float attackSpeedRatio;
    public float criticalDamage;
}
C# KenshinHeroController.cs
// KenshinHeroController.cs
// Contrôleur principal : mouvement, montée en niveau, régénération, cast et passif Battōsai.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class KenshinHeroController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator         animator;
    [SerializeField] private NavMeshAgent     agent;
    [SerializeField] private KenshinCombatVFX combatVfx;
    [SerializeField] private Transform        skillSpawn;
    [SerializeField] private LayerMask        groundMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentMana;
    [SerializeField] private bool  isDead;

    // ── Passif Battōsai ───────────────────────────────────────────────
    private int   consecutiveHits       = 0;
    private float battosaiTimer         = 0f;
    private float battosaiCooldownTimer = 0f;
    private bool  isBattosaiActive      = false;

    // ── Cooldowns ─────────────────────────────────────────────────────
    private float basicCooldown;
    private float skillOneCooldown;
    private float skillTwoCooldown;
    private float ultimateCooldown;

    public RuntimeHeroStats Stats   => stats;
    public float HpPercent          => stats.maxHp   <= 0f ? 0f : currentHp   / stats.maxHp;
    public float ManaPercent        => stats.maxMana <= 0f ? 0f : currentMana / stats.maxMana;
    public bool  IsDead             => isDead;
    public bool  IsBattosaiActive   => isBattosaiActive;

    private void Reset()
    {
        agent    = GetComponent<NavMeshAgent>();
        animator = GetComponentInChildren<Animator>();
    }

    private void Awake()
    {
        if (agent == null) agent = GetComponent<NavMeshAgent>();
        ApplyLevel(heroLevel, true);
    }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        TickBattosai();
        RegenerateResources();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats     = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore)
        {
            currentHp   = stats.maxHp;
            currentMana = stats.maxMana;
        }
        else
        {
            currentHp   = Mathf.Min(currentHp,   stats.maxHp);
            currentMana = Mathf.Min(currentMana, stats.maxMana);
        }
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
        animator.SetBool("IsMoving", true);
    }

    public void BasicAttack(Health target)
    {
        if (target == null || basicCooldown > 0f || isDead) return;
        float bonus  = isBattosaiActive
                     ? heroDefinition.battosaiBleeding.Evaluate(heroLevel) : 0f;
        float damage = stats.physicalAttack + bonus;
        basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
        animator.SetTrigger("BasicAttack");
        target.ApplyPhysicalDamage(damage);
        consecutiveHits++;
        if (consecutiveHits >= 3 && battosaiCooldownTimer <= 0f)
            ActivateBattosai();
    }

    public bool TryCastSkillOne(Transform target)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;
        SpendMana(heroDefinition.skillOne.ManaAtRank(skillOneRank));
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        int   hits      = isBattosaiActive ? 7 : 5;
        float dmgPerHit = heroDefinition.skillOne.DamageAtRank(skillOneRank, stats.physicalAttack);
        agent.isStopped = true;
        transform.LookAt(new Vector3(target.position.x, transform.position.y, target.position.z));
        animator.SetTrigger("CastSkill01");
        combatVfx.PlayRyusosenSlash(skillSpawn.position, target.position, hits);
        target.GetComponent<Health>()?.ApplyPhysicalDamage(dmgPerHit * hits);
        return true;
    }

    public bool TryCastSkillTwo(Vector3 direction)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;
        SpendMana(heroDefinition.skillTwo.ManaAtRank(skillTwoRank));
        skillTwoCooldown = heroDefinition.skillTwo.CooldownAtRank(skillTwoRank);
        float damage    = heroDefinition.skillTwo.DamageAtRank(skillTwoRank, stats.physicalAttack);
        float waveCount = isBattosaiActive ? 2f : 1f;
        agent.isStopped = true;
        transform.forward = (direction - transform.position).normalized;
        animator.SetTrigger("CastSkill02");
        combatVfx.PlayDoryusenWave(transform.position, direction, waveCount);
        Collider[] hits = Physics.OverlapSphere(transform.position, 7f);
        foreach (var col in hits)
        {
            float angle = Vector3.Angle(transform.forward,
                          (col.transform.position - transform.position).normalized);
            if (angle < 60f)
                col.GetComponent<Health>()?.ApplyPhysicalDamage(damage * waveCount);
        }
        return true;
    }

    public bool TryCastUltimate(Transform target)
    {
        if (!CanCast(heroDefinition.ultimate, ultimateRank, ultimateCooldown)) return false;
        SpendMana(heroDefinition.ultimate.ManaAtRank(ultimateRank));
        ultimateCooldown = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        StartCoroutine(AmakakeruRoutine(target));
        return true;
    }

    private IEnumerator AmakakeruRoutine(Transform target)
    {
        agent.isStopped = true;
        combatVfx.SetInvulnerable(true);
        animator.SetTrigger("CastUltimate");
        yield return new WaitForSeconds(0.35f);
        if (target != null)
        {
            Vector3 dashTarget = target.position - transform.forward * 0.8f;
            transform.position = dashTarget;
            agent.Warp(dashTarget);
            float mainDmg = heroDefinition.ultimate.DamageAtRank(ultimateRank, stats.physicalAttack);
            target.GetComponent<Health>()?.ApplyPhysicalDamage(mainDmg);
        }
        combatVfx.PlayAmakakeruStrike(transform.position, 5f + ultimateRank);
        combatVfx.SetInvulnerable(false);
        float waveDmg = ultimateRank == 1 ? 200f : ultimateRank == 2 ? 300f : 420f;
        foreach (var col in Physics.OverlapSphere(transform.position, 5f))
            if (col.transform != target)
                col.GetComponent<Health>()?.ApplyPhysicalDamage(waveDmg);
        yield return new WaitForSeconds(0.4f);
        agent.isStopped = false;
    }

    private void ActivateBattosai()
    {
        isBattosaiActive     = true;
        battosaiTimer        = 5f;
        consecutiveHits      = 0;
        float cd             = Mathf.Lerp(14f, 10f, (heroLevel - 1f) / 14f);
        battosaiCooldownTimer= cd;
        animator.SetBool("BattosaiMode", true);
        combatVfx.SetBattosaiMode(true, heroDefinition.battosaiAtkBonus.Evaluate(heroLevel));
    }

    private void DeactivateBattosai()
    {
        isBattosaiActive = false;
        animator.SetBool("BattosaiMode", false);
        combatVfx.SetBattosaiMode(false, 0f);
    }

    private void TickBattosai()
    {
        if (battosaiCooldownTimer > 0f) battosaiCooldownTimer -= Time.deltaTime;
        if (isBattosaiActive)
        {
            battosaiTimer -= Time.deltaTime;
            if (battosaiTimer <= 0f) DeactivateBattosai();
        }
        if (!isBattosaiActive && battosaiCooldownTimer <= 0f && HpPercent <= 0.35f)
        {
            consecutiveHits = 3;
            ActivateBattosai();
        }
    }

    public void ReceiveDamage(float amount)
    {
        if (isDead) return;
        consecutiveHits = 0;
        currentHp = Mathf.Max(0f, currentHp - amount);
        animator.SetTrigger("Hit");
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, false);
        if (currentHp <= 0f) StartCoroutine(DeathRoutine());
    }

    private IEnumerator DeathRoutine()
    {
        isDead = true;
        agent.isStopped = true;
        DeactivateBattosai();
        animator.SetBool("IsDead", true);
        combatVfx.PlayDeath(transform.position);
        yield return new WaitForSeconds(3f);
        enabled = false;
    }

    private void TickCooldowns()
    {
        basicCooldown    = Mathf.Max(0f, basicCooldown    - Time.deltaTime);
        skillOneCooldown = Mathf.Max(0f, skillOneCooldown - Time.deltaTime);
        skillTwoCooldown = Mathf.Max(0f, skillTwoCooldown - Time.deltaTime);
        ultimateCooldown = Mathf.Max(0f, ultimateCooldown - Time.deltaTime);
    }

    private void RegenerateResources()
    {
        currentHp   = Mathf.Min(stats.maxHp,   currentHp   + stats.hpRegen   * Time.deltaTime);
        currentMana = Mathf.Min(stats.maxMana,  currentMana + stats.manaRegen * Time.deltaTime);
        combatVfx.UpdateHeroState(HpPercent, ManaPercent, agent.velocity.sqrMagnitude > 0.01f);
    }

    private void UpdateAnimator()
    {
        Vector3 planarVelocity = agent.velocity; planarVelocity.y = 0f;
        float effectiveAS = stats.attackSpeed +
            (isBattosaiActive ? heroDefinition.battosaiAtkSpeed.Evaluate(heroLevel) : 0f);
        animator.SetFloat("MoveSpeed",   planarVelocity.magnitude);
        animator.SetFloat("AttackSpeed", effectiveAS);
    }

    private bool CanCast(AbilityRankData ability, int rank, float cd)
    {
        if (heroDefinition == null || isDead || cd > 0f) return false;
        return currentMana >= ability.ManaAtRank(rank);
    }

    private void SpendMana(float amount)
        => currentMana = Mathf.Max(0f, currentMana - amount);
}

using UnityEngine;

public class KenshinCombatVFX : MonoBehaviour

    [SerializeField] private Renderer heroRenderer;
    [SerializeField] private Renderer dashTrailRenderer;
    [SerializeField] private Renderer groundWaveRenderer;
    [SerializeField] private Renderer eyeRenderer;

    private static readonly int MoveBlend        = Shader.PropertyToID("_MoveBlend");
    private static readonly int HpFade           = Shader.PropertyToID("_HpFade");
    private static readonly int ManaGlow         = Shader.PropertyToID("_ManaGlow");
    private static readonly int Dissolve         = Shader.PropertyToID("_Dissolve");
    private static readonly int Radius           = Shader.PropertyToID("_Radius");

    {
        if (heroRenderer == null) return;
        heroRenderer.material.SetFloat(MoveBlend, isMoving ? 1f : 0f);
        heroRenderer.material.SetFloat(HpFade,    1f - hpPercent);
        heroRenderer.material.SetFloat(ManaGlow,  manaPercent);
    }

    {
        if (heroRenderer != null)
            heroRenderer.material.SetFloat(BattosaiIntensity, active ? intensity : 0f);
        if (eyeRenderer != null)
            eyeRenderer.material.SetColor(EyeColor, active ? eyeBattosai : eyeNormal);
        if (kiAuraAmbient != null)
        {
            var main = kiAuraAmbient.main;
            main.startSpeedMultiplier = active ? 2.5f : 1.0f;
            main.startSizeMultiplier  = active ? 1.8f : 1.0f;
        }
    }

    public void PlayRyusosenSlash(Vector3 origin, Vector3 target, int hitCount)
    {
        if (slashCast == null) return;

        slashCast.transform.forward  = (target - origin).normalized;

        {
            amakakeruStrike.transform.position = center;
            amakakeruStrike.Play();
        }
        if (dashTrailRenderer != null)
            dashTrailRenderer.material.SetFloat(Radius, radius);
    }

    public void SetInvulnerable(bool active)
    {
        if (heroRenderer != null)
            heroRenderer.material.SetFloat("_InvulFrame", active ? 1f : 0f);
    }

        {
            deathDissolve.transform.position = position;
            deathDissolve.Play();
        }
        if (heroRenderer != null)
            heroRenderer.material.SetFloat(Dissolve, 1f);
    }

// KenshinKiAura.shader
// URP transparent VFX shader pour l'aura Ki et les trainées de slash.
// Flamme scroll, glow Battosai et fresnel pour rendu gameplay lisible.

Shader "Kenshin/KiAuraEffect"
{
    Properties
    {
        _MainTex        ("Ki Texture", 2D)                  = "white" {}
        _KiColor        ("Ki Color", Color)                 = (0.95, 0.35, 0.10, 0.80)
        _BattosaiColor  ("Battosai Ki Color", Color)        = (1.00, 0.10, 0.05, 1.00)
        _BattosaiBlend  ("Battosai Blend", Range(0,1))      = 0
        _GlowIntensity  ("Glow Intensity", Range(0, 8))     = 2.2
        _FlameSpeed     ("Flame Scroll Speed", Range(0, 8)) = 3.5
        _Distortion     ("UV Distortion", Range(0, 0.25))   = 0.045
        _FresnelPower   ("Fresnel Power", Range(1, 8))      = 2.8
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_MainTex); SAMPLER(sampler_MainTex);

            CBUFFER_START(UnityPerMaterial)
                float4 _MainTex_ST;
                float4 _KiColor;
                float4 _BattosaiColor;
                float  _BattosaiBlend;
                float  _GlowIntensity;
                float  _FlameSpeed;
                float  _Distortion;
                float  _FresnelPower;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float2 uv         : TEXCOORD0;
                float3 normalOS   : NORMAL;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv         : TEXCOORD0;
                float3 normalWS   : TEXCOORD1;
                float3 viewDirWS  : TEXCOORD2;
            };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = TRANSFORM_TEX(IN.uv, _MainTex);
                OUT.normalWS   = normalize(nor.normalWS);
                OUT.viewDirWS  = GetWorldSpaceNormalizeViewDir(pos.positionWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float2 flameUv = IN.uv;
                flameUv.y -= _Time.y * _FlameSpeed * 0.4;
                flameUv.x += sin(_Time.y * _FlameSpeed + IN.uv.y * 10.0) * _Distortion;

                half4 tex     = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, flameUv);
                float fresnel = pow(1.0 - saturate(dot(normalize(IN.normalWS),
                                normalize(IN.viewDirWS))), _FresnelPower);

                half3 kiColor  = lerp(_KiColor.rgb, _BattosaiColor.rgb, _BattosaiBlend);
                half3 emission = kiColor * (_GlowIntensity * tex.a + fresnel * 0.80);

                half4 col;
                col.rgb = tex.rgb * kiColor + emission;
                col.a   = tex.a * lerp(_KiColor.a, _BattosaiColor.a, _BattosaiBlend);
                return col;
            }
            ENDHLSL
        }
    }
}

// KenshinHeroSurface.shader
// Shader héros gameplay-aware : locomotion, mana glow, mode Battōsai, hit, dissolve mort.

Shader "Kenshin/HeroSurface"
{
    Properties
    {
        _BaseMap          ("Base Map", 2D)                        = "white" {}
        _BaseColor        ("Base Color", Color)                   = (1,1,1,1)
        _KiEmission       ("Ki Emission Color", Color)            = (1.0, 0.35, 0.05, 1)
        _BattosaiEmission ("Battosai Emission Color", Color)      = (1.0, 0.08, 0.02, 1)
        _ManaGlow         ("Mana Glow", Range(0,1))               = 1
        _MoveBlend        ("Move Blend", Range(0,1))              = 0
        _HpFade           ("HP Fade", Range(0,1))                 = 0
        _BattosaiIntensity("Battosai Intensity", Range(0,1))      = 0
        _InvulFrame       ("Invul Frame Flash", Range(0,1))       = 0
        _Dissolve         ("Dissolve", Range(0,1))                = 0
        _DissolveSoftness ("Dissolve Softness", Range(0.001,0.2)) = 0.04
        _NoiseScale       ("Noise Scale", Range(1,25))            = 8
    }

    SubShader
    {
        Tags { "RenderType"="TransparentCutout" "Queue"="AlphaTest" "RenderPipeline"="UniversalPipeline" }

        Pass
        {
            Cull Back
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_BaseMap); SAMPLER(sampler_BaseMap);

            CBUFFER_START(UnityPerMaterial)
                float4 _BaseMap_ST;
                float4 _BaseColor;
                float4 _KiEmission;
                float4 _BattosaiEmission;
                float  _ManaGlow;
                float  _MoveBlend;
                float  _HpFade;
                float  _BattosaiIntensity;
                float  _InvulFrame;
                float  _Dissolve;
                float  _DissolveSoftness;
                float  _NoiseScale;
            CBUFFER_END

            struct Attributes
            {
                float4 positionOS : POSITION;
                float3 normalOS   : NORMAL;
                float2 uv         : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float2 uv         : TEXCOORD0;
                float3 worldPos   : TEXCOORD1;
                float3 normalWS   : TEXCOORD2;
            };

            float hash21(float2 p)
            {
                p  = frac(p * float2(123.34, 345.45));
                p += dot(p, p + 34.345);
                return frac(p.x * p.y);
            }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                float battosaiVibration = sin(_Time.y * 18.0 + IN.positionOS.y * 8.0)
                                        * 0.004 * _BattosaiIntensity;
                float moveOffset        = sin(_Time.y * 9.0  + IN.positionOS.y * 6.0)
                                        * 0.003 * _MoveBlend;
                float3 displaced = IN.positionOS.xyz
                                 + IN.normalOS * (moveOffset + battosaiVibration);
                VertexPositionInputs pos = GetVertexPositionInputs(displaced);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = TRANSFORM_TEX(IN.uv, _BaseMap);
                OUT.worldPos   = pos.positionWS;
                OUT.normalWS   = normalize(nor.normalWS);
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                half4 albedo  = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, IN.uv) * _BaseColor;
                float noise   = hash21(floor(IN.worldPos.xz * _NoiseScale));
                float dThresh = saturate(_Dissolve * (1.0 + _DissolveSoftness));
                clip(noise - dThresh);

                float rim         = pow(1.0 - saturate(IN.normalWS.y), 2.5);
                float manaPulse   = (0.45 + 0.55 * sin(_Time.y * 5.0))  * _ManaGlow;
                float kiPulse     = (0.5  + 0.5  * sin(_Time.y * 4.0))  * (1.0 - _BattosaiIntensity);
                float battoPulse  = (0.6  + 0.4  * sin(_Time.y * 12.0)) * _BattosaiIntensity;
                float invulFlash  = (0.5  + 0.5  * sin(_Time.y * 30.0)) * _InvulFrame;
                float lowHpPulse  = (0.25 + 0.75 * sin(_Time.y * 8.0))  * _HpFade;

                half3 emission = _KiEmission.rgb       * (rim * kiPulse)
                               + _BattosaiEmission.rgb * (rim * battoPulse)
                               + half3(1,1,1)          * invulFlash
                               + half3(1,0.1,0)        * lowHpPulse
                               + half3(0.5,0.4,1)      * (rim * manaPulse * 0.3);

                return half4(albedo.rgb + emission, albedo.a);
            }
            ENDHLSL
        }
    }
}

// KenshinDashTrail.shader
// Trainée ultra-rapide pour le dash de l'Amakakeru Ryū no Hirameki.
// Utilisable sur un mesh trail, un quad ou une ribbon VFX.

Shader "Kenshin/DashTrail"
{
    Properties
    {
        _MainColor   ("Main Color", Color)              = (1.0, 0.25, 0.05, 0.90)
        _EdgeColor   ("Edge Color", Color)              = (1.0, 1.0, 0.85, 1.0)
        _ScrollSpeed ("Scroll Speed", Range(0, 10))     = 6.0
        _Dissolve    ("Dissolve", Range(0, 1))          = 0
        _Intensity   ("Intensity", Range(0, 10))        = 3.5
        _Stretch     ("Horizontal Stretch", Range(1,8)) = 3.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha One
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _MainColor;
                float4 _EdgeColor;
                float  _ScrollSpeed;
                float  _Dissolve;
                float  _Intensity;
                float  _Stretch;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = IN.uv;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float stretchedU = IN.uv.x * _Stretch;
                float scan = frac(stretchedU - _Time.y * _ScrollSpeed);
                float band = smoothstep(0.0, 0.15, scan)
                           * (1.0 - smoothstep(0.15, 0.85, scan));
                float edge = pow(saturate(1.0 - abs(IN.uv.y * 2.0 - 1.0)), 3.5);
                float alpha = saturate((1.0 - _Dissolve) * (edge + band * 0.6));
                half3 color = (_MainColor.rgb * edge + _EdgeColor.rgb * band) * _Intensity;
                return half4(color, alpha * _MainColor.a);
            }
            ENDHLSL
        }
    }

// DoryusenGroundWave.shader
// Zone d'onde au sol pour le Doryūsen : arc de fracture énergétique.
// Pulsation en arc vers l'avant, lisible en teamfight MOBA.

Shader "Kenshin/DoryusenGroundWave"
{
    Properties
    {
        _Color      ("Wave Color", Color)               = (0.85, 0.15, 0.05, 0.40)
        _EdgeColor  ("Edge Color", Color)               = (1.0, 0.65, 0.10, 0.95)
        _Radius     ("Wave Radius", Float)              = 7.0
        _RingWidth  ("Ring Width", Range(0.05, 3))      = 0.55
        _PulseSpeed ("Pulse Speed", Range(0, 6))        = 2.5
        _ArcAngle   ("Arc Angle (deg)", Range(10, 180)) = 60.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _EdgeColor;
                float  _Radius;
                float  _RingWidth;
                float  _PulseSpeed;
                float  _ArcAngle;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = IN.uv * 2.0 - 1.0;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float dist  = length(IN.uv) * _Radius;
                float pulse = 0.90 + 0.10 * sin(_Time.y * _PulseSpeed * 6.2831);
                float ring  = smoothstep(_Radius, _Radius - _RingWidth, dist * pulse);
                float edge  = smoothstep(_Radius, _Radius - (_RingWidth * 0.3), dist * pulse);
                float angleRad = radians(_ArcAngle);
                float angle    = atan2(IN.uv.x, IN.uv.y);
                float arcMask  = step(abs(angle), angleRad);
                half3 color = lerp(_EdgeColor.rgb, _Color.rgb, ring);
                float alpha = saturate(ring * _Color.a + edge * _EdgeColor.a * 0.5) * arcMask;
                return half4(color, alpha);
            }
