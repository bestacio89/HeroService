Silhouette & Morphology

Massively built divine warrior — ~2.05m, broad beyond reason, every inch of him built for war. Thick neck, barrel chest, arms that swing Mjölnir like a toy. Norse myth Thor is not the lean Marvel figure — he is the crushing weight of a thunderstorm made flesh: immense, unstoppable, loud. Short dark-auburn beard, fierce blue storm-grey eyes that crackle with lightning when enraged. Walks like the ground owes him deference. His mass is pure aggression in motion.

Armor — Asgardian Warrior

Battle-worn Norse plate: deep storm-grey iron breastplate engraved with Valknut and Vegvísir (Norse compass) symbols that pulse with embedded lightning channels. Heavy fur-trimmed pauldrons — bear pelts over iron caps. Chainmail underlayer visible at joints. Norse leather bracers with rune-carved iron studs. Heavy iron greaves, thick battle boots. A deep storm-blue short cloak billowing with electrical charge. The Belt of Strength (Megingjörð) at the waist — a wide leather war belt glowing faintly gold, doubling his might. Iron war helm with a simple noseguard — not fully face-covering; his expression of fury must be visible.

Sacred Weapon — Mjölnir

The war hammer Mjölnir — short-handled (historically accurate), devastatingly heavy head engraved with Norse knotwork. The head crackles with arcing blue-white electricity at all times. When thrown it leaves a tumbling end-over-end arc of lightning bolts across the air, strikes with explosive thunder impact, and returns to Thor's outstretched hand via a crackling electric recall tether — the hammer always comes back. A second weapon: the iron war axe Járngreipr at the hip, used in the basic attack combo's final strike when Mjölnir is away during a throw.

Storm Charge — The Core Resource

Thor does not use mana. He uses Storm Charge — electrical energy that builds as he fights. Every melee hit, every lightning strike landed, every kill generates Storm Charge. At max charge (100%) his armor crackles visibly, his eyes glow white, the sky above the arena visibly darkens, and he enters Storm Fury — a passive overclock state that empowers all abilities. The Ultimate requires maximum Storm Charge, drains it entirely, and calls the full fury of Midgard's storms down on the battlefield in one catastrophic event.

System Proposal

Thor is a pure Fighter/Warrior — the highest physical attack in the roster, no healing, no utility, total aggression. He uses Storm Charge (0–100), a resource that only builds through combat: basic attacks, ability hits, and kills all generate charge. It does not regenerate safely out of combat — he must stay in the fight.

At 100 Storm Charge he enters Storm Fury, passively overclock state: every ability costs less, attack speed surges, and lightning arcs secondary-hit nearby enemies on every swing. The Ultimate requires full charge to activate — so managing when to spend Fury versus when to save it for Ragnarök Strike defines his moment-to-moment game.

public enum HeroRole { Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin }

[Serializable]
public struct ScaledStat
{
    public float level1, level15;
    public ScaledStat(float l1, float l15) { level1 = l1; level15 = l15; }
    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return level == 1 ? level1 : Mathf.Lerp(level1, level15, (level - 1f) / 14f);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] chargeCost;
    public float[] cooldown;
    public float   scalingPhysical;

    public AbilityRankData(string n, float[] dmg, float[] cost, float[] cd, float scale)
    { abilityName = n; baseDamage = dmg; chargeCost = cost; cooldown = cd; scalingPhysical = scale; }

    public float DamageAtRank   (int r, float atq) => baseDamage[Mathf.Clamp(r-1,0,baseDamage.Length-1)] + atq * scalingPhysical;
    public float ChargeCostAtRank(int r)          => chargeCost[Mathf.Clamp(r-1,0,chargeCost.Length-1)];
    public float CooldownAtRank  (int r)          => cooldown  [Mathf.Clamp(r-1,0,cooldown.Length-1)];
}

// ThorController.cs
// Main controller: Storm Charge float (builds in combat, passive slow regen),
// Storm Fury bool + 15s timer + kill extension, 3-hit combo state machine,
// hammer-in-flight flag (fights with axe during throw), Mjölnir return arc,
// Thunder Stride leap + sustained lightning field, Ragnarök untargetable air phase
// + crater AoE + 6 pillar strikes + optional Rank III EMP stun.

using UnityEngine;
using UnityEngine.AI;
using System.Collections;
using System.Collections.Generic;

[RequireComponent(typeof(NavMeshAgent))]
public class ThorController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator       animator;
    [SerializeField] private NavMeshAgent   agent;
    [SerializeField] private ThorCombatVFX  combatVfx;
    [SerializeField] private Transform       hammerGrip;
    [SerializeField] private Transform       axeGrip;
    [SerializeField] private LayerMask       enemyMask;
    [SerializeField] private LayerMask       wallMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentCharge;
    [SerializeField] private bool  isDead;
    [SerializeField] private bool  stormFuryActive;
    [SerializeField] private bool  hammerInFlight;
    [SerializeField] private bool  isUntargetable; // during Ragnarök air phase

    private float furyTimer;
    private float basicCooldown, skillOneCooldown, skillTwoCooldown, ultimateCooldown;

    // 3-hit combo state
    private int   comboStep       = 0;
    private float comboResetTimer = 0f;
    private const float COMBO_RESET = 1.8f;

    public float HpPercent     => stats.maxHp          > 0f ? currentHp     / stats.maxHp          : 0f;
    public float ChargePercent => stats.maxStormCharge > 0f ? currentCharge / stats.maxStormCharge : 0f;
    public bool  StormFury     => stormFuryActive;
    public bool  IsDead        => isDead;

    // Fury halves ability costs when active
    private float FuryCost(float baseCost)
        => stormFuryActive ? baseCost * (1f - heroDefinition.furyCostReduction) : baseCost;

    private float GetEffectiveAttackSpeed()
        => stormFuryActive ? stats.attackSpeed * (1f + heroDefinition.furyAttackSpeedBonus) : stats.attackSpeed;

    private void Reset() { agent = GetComponent<NavMeshAgent>(); animator = GetComponentInChildren<Animator>(); }
    private void Awake() { if (agent == null) agent = GetComponent<NavMeshAgent>(); ApplyLevel(heroLevel, true); }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        RegenerateResources();
        TickFury();
        TickComboReset();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats     = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore) { currentHp = stats.maxHp; currentCharge = 0f; }
        else currentHp = Mathf.Min(currentHp, stats.maxHp);
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead || isUntargetable) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
    }

    // ── BASIC ATTACK — 3-hit combo ────────────────────────────────────────
    public void BasicAttack(Health primaryTarget)
    {
        if (isDead || isUntargetable || basicCooldown > 0f || primaryTarget == null) return;

        comboResetTimer = COMBO_RESET;
        basicCooldown   = 1f / Mathf.Max(0.1f, GetEffectiveAttackSpeed() * stats.attackSpeedRatio);
        float atq = stats.physicalAttack * (1f + stats.armorPen);

        switch (comboStep)
        {
            case 0: // Iron Fist
                ApplyHit(primaryTarget, atq * 0.90f, false);
                animator.SetTrigger("Attack1");
                combatVfx.PlayFistImpact(primaryTarget.transform.position);
                break;
            case 1: // Járngreipr axe sweep — AoE cone
                animator.SetTrigger("Attack2");
                combatVfx.PlayAxeSlash(transform.position, transform.forward);
                foreach (var col in Physics.OverlapSphere(transform.position, stats.basicAttackRange, enemyMask))
                {
                    Vector3 dir = (col.transform.position - transform.position).normalized;
                    if (Vector3.Angle(transform.forward, dir) <= 60f)
                        ApplyHit(col.GetComponent<Health>(), atq * 1.10f, false);
                }
                break;
            case 2: // Mjölnir overhead slam — ground shockwave, stun primary
                animator.SetTrigger("Attack3");
                combatVfx.PlayHammerSlam(primaryTarget.transform.position);
                ApplyHit(primaryTarget, atq * 1.50f, false);
                primaryTarget.GetComponent<CrowdControlReceiver>()?.ApplyStun(0.4f);
                // shockwave ring hits nearby
                foreach (var col in Physics.OverlapSphere(primaryTarget.transform.position, 2.5f, enemyMask))
                    if (col.GetComponent<Health>() != primaryTarget) ApplyHit(col.GetComponent<Health>(), atq * 0.55f, false);
                break;
        }
        comboStep = (comboStep + 1) % 3;
    }

    // ── SKILL 1 — Mjölnir Throw ───────────────────────────────────────────
    public bool TryCastSkillOne(Vector3 direction)
    {
        if (isDead || isUntargetable || skillOneCooldown > 0f || hammerInFlight) return false;
        float cost = FuryCost(heroDefinition.skillOne.ChargeCostAtRank(skillOneRank));
        if (currentCharge < cost) return false;
        SpendCharge(cost);
        skillOneCooldown = heroDefinition.skillOne.CooldownAtRank(skillOneRank);
        animator.SetTrigger("ThrowHammer");
        StartCoroutine(MjölnirThrowRoutine(direction.normalized, skillOneRank));
        return true;
    }

    private IEnumerator MjölnirThrowRoutine(Vector3 dir, int rank)
    {
        hammerInFlight = true;
        float throwRange  = 12f;
        float throwSpeed  = 22f;
        float travelTime  = throwRange / throwSpeed;
        float throwDmg    = heroDefinition.skillOne.DamageAtRank(rank, stats.physicalAttack);
        float returnDmg   = throwDmg * 0.5f;
        float arcDmg      = 100f + rank * 30f + stats.magicPower * (0.35f + rank * 0.05f);

            {
                var h = col.GetComponent<Health>();
                if (h == null || already.Contains(h)) continue;
                float d = Vector3.Distance(pos, col.transform.position);
                if (d < best) { best = d; next = h; }

            if (next == null) break;
            combatVfx.PlayLightningArc(pos, next.transform.position);
            ApplyHit(next, arcDmg, true);
            already.Add(next);
            pos = next.transform.position;
            arced++;
            yield return new WaitForSeconds(0.06f);

                float fullFlash   = saturate((_ChargePercent - 0.88) * 8.0) * (0.5 + 0.5 * sin(_Time.y * _PulseSpeed * 3.0));
                // Fury overclock — everything goes gold-electric
                float furyPulse   = (0.4 + 0.6 * sin(_Time.y * _PulseSpeed * 2.2)) * _FuryActive;

                half3 runeEmit  = _RuneColor.rgb  * _RuneIntensity * (runeband * chargePulse + fresnel * chargePulse * 0.4 + fullFlash);
                half3 furyEmit  = _FuryColor.rgb  * _FuryIntensity * furyPulse * (fresnel * 0.6 + runeband * 0.4);

                // Darken when HP drops
                albedo.rgb *= (1.0 - _HpFade * 0.4);

                return half4(albedo.rgb + runeEmit + furyEmit, albedo.a);
            }
            ENDHLSL
        }
    }
}

// MjölnirCharge.shader
// Applied to the Mjölnir hammer head emission zone.
// _ChargePercent : dim knotwork → blazing blue-white arc field
// _FuryActive    : hammer head becomes a supernova of electric gold
// Vertex arc bolts: UV-driven spark tendrils that intensify with charge.

Shader "Thor/MjölnirCharge"
{
    Properties
    {
        _BaseMap       ("Base Map", 2D)               = "white" {}
        _BaseColor     ("Iron Color", Color)           = (0.28, 0.30, 0.35, 1.0)
        _ArcColor      ("Arc Color", Color)            = (0.50, 0.75, 1.00, 1.0)
        _FuryColor     ("Fury Arc Color", Color)       = (1.00, 0.88, 0.20, 1.0)
        _ChargePercent ("Charge %",    Range(0,1))     = 0.0
        _FuryActive    ("Fury Active", Range(0,1))     = 0.0
        _Intensity     ("Arc Intensity",   Range(0,14)) = 4.0
        _ArcSpeed      ("Arc Speed",       Range(0,12)) = 5.0
        _ArcDensity    ("Arc Density",     Range(2,20)) = 8.0
        _FresnelPow    ("Fresnel Power",   Range(1,8))  = 2.0
        _Dissolve      ("Dissolve",        Range(0,1))  = 0.0
    }

    SubShader
    {
        Tags { "RenderType"="Opaque" "RenderPipeline"="UniversalPipeline" }

        Pass
        {
            Cull Back
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            TEXTURE2D(_BaseMap); SAMPLER(sampler_BaseMap);

            CBUFFER_START(UnityPerMaterial)
                float4 _BaseMap_ST;
                float4 _BaseColor;
                float4 _ArcColor;
                float4 _FuryColor;
                float  _ChargePercent;
                float  _FuryActive;
                float  _Intensity;
                float  _ArcSpeed;
                float  _ArcDensity;
                float  _FresnelPow;
                float  _Dissolve;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float3 normalOS : NORMAL; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; float3 normalWS : TEXCOORD1; float3 viewDirWS : TEXCOORD2; float3 worldPos : TEXCOORD3; };

            float hash21(float2 p) { p = frac(p * float2(127.1, 311.7)); p += dot(p, p + 34.345); return frac(p.x * p.y); }

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                VertexPositionInputs pos = GetVertexPositionInputs(IN.positionOS.xyz);
                VertexNormalInputs   nor = GetVertexNormalInputs(IN.normalOS);
                OUT.positionCS = pos.positionCS;
                OUT.uv         = TRANSFORM_TEX(IN.uv, _BaseMap);
                OUT.normalWS   = normalize(nor.normalWS);
                OUT.viewDirWS  = GetWorldSpaceNormalizeViewDir(pos.positionWS);
                OUT.worldPos   = pos.positionWS;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                // Death — Mjölnir stays (Dissolve stays at 0 on death for the hammer)
                float noiseD = hash21(floor(IN.worldPos.xz * 10.0));
                clip(noiseD - saturate(_Dissolve * 1.05));

                half4 albedo  = SAMPLE_TEXTURE2D(_BaseMap, sampler_BaseMap, IN.uv) * _BaseColor;
                float fresnel = pow(1.0 - saturate(dot(IN.normalWS, IN.viewDirWS)), _FresnelPow);

                // Arc tendrils across hammer face — animated with charge
                float2 arcUv  = IN.uv;
                arcUv.x += sin(_Time.y * _ArcSpeed + IN.uv.y * _ArcDensity) * 0.04;
                arcUv.y += cos(_Time.y * _ArcSpeed * 0.7 + IN.uv.x * _ArcDensity * 0.8) * 0.04;
                float arc = frac(arcUv.x * _ArcDensity * 0.5 + arcUv.y * _ArcDensity);
                arc = pow(smoothstep(0.0, 0.35, arc) * (1.0 - smoothstep(0.35, 0.8, arc)), 1.5);

                float chargeMult = _ChargePercent;
                float fullFlash  = saturate((_ChargePercent - 0.85) * 6.67) * (0.6 + 0.4 * sin(_Time.y * _ArcSpeed * 2.5));
                float furyBoost  = (0.5 + 0.5 * sin(_Time.y * _ArcSpeed * 1.8)) * _FuryActive;

                half3 arcEmit  = _ArcColor.rgb   * _Intensity * (arc * chargeMult + fresnel * chargeMult * 0.5 + fullFlash * 0.8);
                half3 furyEmit = _FuryColor.rgb  * _Intensity * furyBoost * (arc * 1.5 + fresnel);

                return half4(albedo.rgb + arcEmit + furyEmit, albedo.a);
            }
            ENDHLSL
        }
    }
}
HLSL LightningFieldGround.shader
// LightningFieldGround.shader
// Applied to a disc quad at the Thunder Stride landing point.
// Crackling electric-blue fill with jagged ring edge — communicates DANGER, not healing.
// Visually opposite to the Pharmakeia Aura: harsh, jagged, electric.
// _FuryActive doubles the radius (handled CPU-side on _Radius set).

Shader "Thor/LightningFieldGround"
{
    Properties
    {
        _CoreColor  ("Core Color",  Color)          = (0.15, 0.35, 0.90, 0.35)
        _EdgeColor  ("Edge Color",  Color)          = (0.70, 0.90, 1.00, 0.95)
        _BoltColor  ("Bolt Color",  Color)          = (1.00, 1.00, 1.00, 1.00)
        _Radius     ("Radius",      Float)          = 5.0
        _BoltDensity("Bolt Density", Range(2,20))   = 8.0
        _BoltSpeed  ("Bolt Speed",   Range(0,10))   = 6.5
        _PulseSpeed ("Pulse Speed",  Range(0,6))    = 3.5
        _FadeOut    ("Fade Out",     Range(0,1))    = 0.0
    }

    SubShader
    {
        Tags { "RenderType"="Transparent" "Queue"="Transparent" "RenderPipeline"="UniversalPipeline" }
        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            CBUFFER_START(UnityPerMaterial)
                float4 _CoreColor;
                float4 _EdgeColor;
                float4 _BoltColor;
                float  _Radius;
                float  _BoltDensity;
                float  _BoltSpeed;
                float  _PulseSpeed;
                float  _FadeOut;
            CBUFFER_END

            struct Attributes { float4 positionOS : POSITION; float2 uv : TEXCOORD0; };
            struct Varyings   { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; };

            Varyings vert(Attributes IN)
            {
                Varyings OUT;
                OUT.positionCS = GetVertexPositionInputs(IN.positionOS.xyz).positionCS;
                OUT.uv = IN.uv * 2.0 - 1.0;
                return OUT;
            }

            half4 frag(Varyings IN) : SV_Target
            {
                float dist = length(IN.uv);
                float pulse = 0.85 + 0.15 * sin(_Time.y * _PulseSpeed * 6.2831);

                // Zone fill
                float fill = smoothstep(1.0, 0.80, dist) * pulse;

                // Jagged bolt pattern — unlike the smooth healing ring
                float angle = atan2(IN.uv.y, IN.uv.x);
                float boltWave = 0.5 + 0.5 * sin(angle * _BoltDensity + _Time.y * _BoltSpeed);
                float bolt = pow(boltWave, 5.0) * smoothstep(0.85, 0.65, dist) * smoothstep(0.30, 0.65, dist);

                // Outer edge crackle ring
                float ring = smoothstep(1.0, 0.90, dist) * smoothstep(0.82, 1.0, dist) * pulse;

                float fade   = saturate(1.0 - _FadeOut);
                half3 color  = _CoreColor.rgb * fill + _EdgeColor.rgb * ring + _BoltColor.rgb * bolt;
                float alpha  = (fill * _CoreColor.a * 0.5 + bolt * 0.7 + ring * _EdgeColor.a) * fade;

                return half4(color, saturate(alpha));
            }
            ENDHLSL
        }
    }
}

                half3 color  = _CraterColor.rgb * scorch + _EmberColor.rgb * ember * pulse + _GoldEdge.rgb * gold;
                float alpha  = (scorch * _CraterColor.a + ember * _EmberColor.a * pulse + gold * _GoldEdge.a * 0.55) * fade;
