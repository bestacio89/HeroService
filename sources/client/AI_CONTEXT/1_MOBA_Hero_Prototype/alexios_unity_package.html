Silhouette & Morphology

Powerfully built Spartan mercenary — tall, muscular, broad-shouldered. ~1.90m. Scarred, sun-bronzed skin from years of Aegean warfare. Thick dark beard, intense amber-gold eyes reflecting Isu heritage. Moves with a warrior's confident aggression — heavy strikes, explosive lunges, no wasted motion. Carries the weight of a demigod in every stance.

Armor — Spartan Mercenary

Layered Greek warrior armor: bronze cuirass with engraved Spartan lambda (Λ) on the breastplate, leather pteruges (hanging strips) at hips and shoulders, greaves on the shins, sandaled boots. Worn crimson-red cloak over the left shoulder. Crested Spartan helmet optional (off during standard play — worn for ultimate). Brown leather straps and bracers. Battle-worn with dents and scratches.

Iconic Weapon — Spear of Leonidas

The broken Spear of Leonidas — legendary Isu artifact. A short, gleaming blade mounted on a broken haft — not a full spear but a half-spear used as a blade-on-a-stick. Gold Isu glyphs running up the shaft, crackling with amber energy. The broken tip pulses with otherworldly power during abilities. Primary weapon — used for both melee strikes and the iconic energy-channel abilities that define Alexios's kit.

Secondary — Spartan Shield & Sword

Hoplon shield — round bronze Greek shield with Spartan lambda embossed at center, red rim, worn leather back grip. Xiphos shortsword at the hip for close-quarters quick strikes. The shield doubles as an offensive tool: barge, bash, and the Spartan Kick (the iconic "THIS IS SPARTA" boot that sends enemies flying). Both weapons active in the basic attack combo chain.

System Proposal

Alexios is a bruiser Fighter/Assassin — high HP for a melee fighter, excellent physical defense from the hoplon shield, outstanding burst damage from Isu-powered abilities. He uses an Adrenaline resource that regenerates continuously and surges on kills — rewarding aggressive play.

His iconic Spear of Leonidas drives all three active abilities. His basic attack chains spear, sword, and shield in a 3-hit combo with the third hit always being the Spartan Shield Bash, which briefly stuns. The passive Isu Bloodline grants escalating power at low HP — a true cornered warrior becomes more dangerous.

// HeroDefinition.cs
// ScriptableObject — MOBA 15-level hero data.
// Extended for Alexios: Adrenaline resource, armor penetration, Isu passive thresholds.

using UnityEngine;
using System;

[CreateAssetMenu(menuName = "Game/Hero Definition", fileName = "HeroDefinition")]
public class HeroDefinition : ScriptableObject
{
    [Header("Identity")]
    public string   heroId      = "alexios";
    public string   displayName = "Alexios of Sparta";
    public HeroRole role        = HeroRole.Fighter;

    [Header("Core progression — Level 1 to 15")]
    public ScaledStat maxHp           = new(3600f, 7400f);
    public ScaledStat hpRegen          = new(11.0f, 24.0f);
    public ScaledStat adrenalineRegen  = new(12.0f, 22.0f);
    public ScaledStat physicalAttack   = new(175f,  360f);
    public ScaledStat physicalDefense  = new(48f,   110f);
    public ScaledStat magicDefense     = new(30f,   72f);
    public ScaledStat attackSpeed      = new(0.95f, 1.20f);
    public ScaledStat armorPenetration = new(0.08f, 0.22f); // 8% → 22%

    [Header("Fixed properties")]
    public float moveSpeed         = 245f;
    public float basicAttackRange  = 2.8f;
    public float attackSpeedRatio  = 1.0f;
    public float maxAdrenaline     = 100f;
    public float killAdrenalineBonus = 30f;
    public float shieldBashAdrenaline = 15f;
    public float spartanKickAdrenaline = 20f;

    [Header("Isu Bloodline Passive")]
    public float isuHpThreshold     = 0.40f; // 40% HP
    public float isuAttackBonus     = 0.18f; // +18% physical attack
    public float isuCdReduction     = 0.20f; // cooldowns tick 20% faster
    public float isuPenMultiplier   = 2.0f;  // double armor penetration

    [Header("Ability ranks")]
    public AbilityRankData basicAttack = new("Spartan Combo",
        new[] { 0f }, new[] { 0f }, new[] { 0f }, 0f);

    public AbilityRankData skillOne = new("Spear Throw",
        new[] { 280f, 350f, 420f, 490f },
        new[] { 28f, 28f, 28f, 28f },
        new[] { 9f, 8f, 7f, 6f }, 0.50f);

    public AbilityRankData skillTwo = new("Bull Rush",
        new[] { 220f, 280f, 340f, 400f },
        new[] { 24f, 24f, 24f, 24f },
        new[] { 12f, 11f, 10f, 9f }, 0.60f);

    public AbilityRankData ultimate = new("Wrath of the Isu",
        new[] { 800f, 1100f, 1450f },
        new[] { 100f, 100f, 100f }, // always costs full bar
        new[] { 95f, 85f, 75f }, 0.70f);

    public RuntimeHeroStats CreateRuntime(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return new RuntimeHeroStats
        {
            level            = level,
            maxHp            = maxHp.Evaluate(level),
            hpRegen          = hpRegen.Evaluate(level),
            adrenalineRegen  = adrenalineRegen.Evaluate(level),
            physicalAttack   = physicalAttack.Evaluate(level),
            physicalDefense  = physicalDefense.Evaluate(level),
            magicDefense     = magicDefense.Evaluate(level),
            attackSpeed      = attackSpeed.Evaluate(level),
            armorPenetration = armorPenetration.Evaluate(level),
            moveSpeed        = moveSpeed,
            basicAttackRange = basicAttackRange,
            attackSpeedRatio = attackSpeedRatio,
            maxAdrenaline    = maxAdrenaline
        };
    }
}

public enum HeroRole { Tank, Fighter, Assassin, Mage, Marksman, Support, MageAssassin }

[Serializable]
public struct ScaledStat
{
    public float level1, level15;
    public ScaledStat(float l1, float l15) { level1 = l1; level15 = l15; }
    public float Evaluate(int level)
    {
        level = Mathf.Clamp(level, 1, 15);
        return level == 1 ? level1 : Mathf.Lerp(level1, level15, (level - 1f) / 14f);
    }
}

[Serializable]
public class AbilityRankData
{
    public string  abilityName;
    public float[] baseDamage;
    public float[] adrenalineCost;
    public float[] cooldown;
    public float   scalingPhysical;

    public AbilityRankData(string n, float[] dmg, float[] cost, float[] cd, float scale)
    { abilityName = n; baseDamage = dmg; adrenalineCost = cost; cooldown = cd; scalingPhysical = scale; }

    public float DamageAtRank    (int r, float atq) => baseDamage     [Mathf.Clamp(r-1,0,baseDamage.Length-1)]     + atq * scalingPhysical;
    public float AdrenalineAtRank(int r)          => adrenalineCost  [Mathf.Clamp(r-1,0,adrenalineCost.Length-1)];
    public float CooldownAtRank  (int r)          => cooldown         [Mathf.Clamp(r-1,0,cooldown.Length-1)];
}

[Serializable]
public struct RuntimeHeroStats
{
    public int   level;
    public float maxHp, hpRegen;
    public float adrenalineRegen, maxAdrenaline;
    public float physicalAttack, physicalDefense, magicDefense;
    public float attackSpeed, moveSpeed, basicAttackRange, attackSpeedRatio;
    public float armorPenetration;
}
C# AlexiosController.cs
// AlexiosController.cs
// Main controller: 3-hit combo state machine, Adrenaline resource,
// Isu Bloodline passive, Spear Throw with return arc,
// Bull Rush wall-slam detection, Staff of Hermes ultimate gate (full adrenaline).

using UnityEngine;
using UnityEngine.AI;
using System.Collections;

[RequireComponent(typeof(NavMeshAgent))]
public class AlexiosController : MonoBehaviour
{
    [Header("Data")]
    [SerializeField] private HeroDefinition heroDefinition;
    [SerializeField, Range(1, 15)] private int heroLevel = 1;

    [Header("Ability Ranks")]
    [SerializeField, Range(1, 4)] private int skillOneRank = 1;
    [SerializeField, Range(1, 4)] private int skillTwoRank = 1;
    [SerializeField, Range(1, 3)] private int ultimateRank = 1;

    [Header("Scene References")]
    [SerializeField] private Animator         animator;
    [SerializeField] private NavMeshAgent     agent;
    [SerializeField] private AlexiosCombatVFX combatVfx;
    [SerializeField] private Transform         spearTip;
    [SerializeField] private Transform         staffMount;
    [SerializeField] private LayerMask         enemyMask;
    [SerializeField] private LayerMask         wallMask;

    [Header("Runtime")]
    [SerializeField] private RuntimeHeroStats stats;
    [SerializeField] private float currentHp;
    [SerializeField] private float currentAdrenaline;
    [SerializeField] private bool  isDead;

    // 3-hit combo state
    private int   comboStep      = 0; // 0=sword, 1=spear, 2=shield bash
    private float comboResetTimer = 0f;
    private const float COMBO_RESET_TIME = 1.6f;

    private float basicCooldown, skillOneCooldown, skillTwoCooldown, ultimateCooldown;
    private bool  isuPassiveActive;
    private bool  isUltimateChanneling;

    public float HpPercent         => stats.maxHp        > 0f ? currentHp         / stats.maxHp        : 0f;
    public float AdrenalinePercent  => stats.maxAdrenaline > 0f ? currentAdrenaline / stats.maxAdrenaline : 0f;
    public bool  IsuActive          => isuPassiveActive;
    public bool  IsDead             => isDead;

    private void Reset() { agent = GetComponent<NavMeshAgent>(); animator = GetComponentInChildren<Animator>(); }
    private void Awake() { if (agent == null) agent = GetComponent<NavMeshAgent>(); ApplyLevel(heroLevel, true); }

    private void Update()
    {
        if (isDead || heroDefinition == null) return;
        TickCooldowns();
        RegenerateResources();
        CheckIsuPassive();
        TickComboReset();
        UpdateAnimator();
    }

    public void ApplyLevel(int newLevel, bool fullRestore = false)
    {
        heroLevel = Mathf.Clamp(newLevel, 1, 15);
        stats = heroDefinition.CreateRuntime(heroLevel);
        agent.speed = stats.moveSpeed / 100f;
        if (fullRestore) { currentHp = stats.maxHp; currentAdrenaline = 0f; }
        else currentHp = Mathf.Min(currentHp, stats.maxHp);
    }

    public void MoveTo(Vector3 worldPoint)
    {
        if (isDead) return;
        agent.isStopped = false;
        agent.SetDestination(worldPoint);
    }

    // ── BASIC ATTACK — 3-hit combo ────────────────────────────────────────
    public void BasicAttack(Health target)
    {
        if (isDead || basicCooldown > 0f || target == null) return;

        comboResetTimer = COMBO_RESET_TIME;
        float atq = GetEffectiveAttack();

        switch (comboStep)
        {
            case 0: // Xiphos sword slash
                basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
                target.ApplyPhysicalDamage(ApplyPen(atq * 0.70f));
                animator.SetTrigger("Attack1");
                combatVfx.PlaySwordSlash(spearTip.position);
                break;
            case 1: // Spear thrust
                basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
                target.ApplyPhysicalDamage(ApplyPen(atq * 1.10f));
                animator.SetTrigger("Attack2");
                combatVfx.PlaySpearThrust(spearTip.position);
                break;
            case 2: // Hoplon shield bash — stun + adrenaline gain
                basicCooldown = 1f / Mathf.Max(0.1f, stats.attackSpeed * stats.attackSpeedRatio);
                target.ApplyPhysicalDamage(ApplyPen(atq * 1.40f));
                target.GetComponent<CrowdControlReceiver>()?.ApplyStun(0.5f);
                GainAdrenaline(heroDefinition.shieldBashAdrenaline);
                animator.SetTrigger("Attack3");
                combatVfx.PlayShieldBash(spearTip.position);
                break;
        }
        comboStep = (comboStep + 1) % 3;
    }

    // ── SKILL 1 — Spear Throw: pierce line + Isu Burn + return arc ─────────
    public bool TryCastSkillOne(Vector3 direction)
    {
        if (!CanCast(heroDefinition.skillOne, skillOneRank, skillOneCooldown)) return false;
        SpendAdrenaline(heroDefinition.skillOne.AdrenalineAtRank(skillOneRank));
        skillOneCooldown = GetCooldown(heroDefinition.skillOne.CooldownAtRank(skillOneRank));
        agent.isStopped = true;
        animator.SetTrigger("SpearThrow");
        StartCoroutine(SpearThrowRoutine(direction, skillOneRank));
        return true;
    }

    private IEnumerator SpearThrowRoutine(Vector3 dir, int rank)
    {
        float range     = 14f;
        float burnDmg   = 50f + rank * 15f;     // per second Isu burn
        float burnTime  = 3f;
        float impactDmg = heroDefinition.skillOne.DamageAtRank(rank, GetEffectiveAttack());
        float returnDmg = impactDmg * 0.5f;

        combatVfx.PlaySpearThrowFx(spearTip.position, dir.normalized * range);

        // Forward pierce — raycastAll equivalent
        var hits = Physics.RaycastAll(spearTip.position, dir.normalized, range, enemyMask);
        System.Array.Sort(hits, (a, b) => a.distance.CompareTo(b.distance));
        foreach (var h in hits)
        {
            var hp = h.collider.GetComponent<Health>();
            if (hp != null)
            {
                hp.ApplyPhysicalDamage(ApplyPen(impactDmg));
                hp.GetComponent<StatusEffectReceiver>()?.ApplyBurn(burnDmg, burnTime);
            }
        }

        yield return new WaitForSeconds(0.55f); // spear flight time out

        // Return arc — hits enemies near the return path
        Vector3 returnOrigin = spearTip.position + dir.normalized * range;
        combatVfx.PlaySpearReturnFx(returnOrigin, spearTip.position);
        foreach (var col in Physics.OverlapCapsule(returnOrigin, spearTip.position, 1.2f, enemyMask))
            col.GetComponent<Health>()?.ApplyPhysicalDamage(ApplyPen(returnDmg));

        yield return new WaitForSeconds(0.3f);
        agent.isStopped = false;
    }

    // ── SKILL 2 — Bull Rush / Spartan Kick: charge + knockback + wall slam ─
    public bool TryCastSkillTwo(Vector3 direction)
    {
        if (!CanCast(heroDefinition.skillTwo, skillTwoRank, skillTwoCooldown)) return false;
        SpendAdrenaline(heroDefinition.skillTwo.AdrenalineAtRank(skillTwoRank));
        skillTwoCooldown = GetCooldown(heroDefinition.skillTwo.CooldownAtRank(skillTwoRank));
        animator.SetTrigger("BullRush");
        StartCoroutine(BullRushRoutine(direction, skillTwoRank));
        return true;
    }

    private IEnumerator BullRushRoutine(Vector3 dir, int rank)
    {
        agent.isStopped = true;
        dir = new Vector3(dir.x, 0, dir.z).normalized;
        float chargeRange = 7f;
        float chargeSpeed = 16f;
        float elapsed = 0f;
        float duration = chargeRange / chargeSpeed;
        bool  kicked = false;

        combatVfx.PlayBullRushCharge(transform.position, dir);

        while (elapsed < duration && !kicked)
        {
            transform.Translate(dir * chargeSpeed * Time.deltaTime, Space.World);
            foreach (var col in Physics.OverlapSphere(transform.position, 1.6f, enemyMask))
            {
                var hp = col.GetComponent<Health>();
                if (hp == null) continue;

                float kickDmg = heroDefinition.skillTwo.DamageAtRank(rank, GetEffectiveAttack());
                hp.ApplyPhysicalDamage(ApplyPen(kickDmg));
                animator.SetTrigger("SpartanKick");
                combatVfx.PlaySpartanKick(col.transform.position);

                float knockDist = 4.5f + rank * 0.5f;
                var   cc = col.GetComponent<CrowdControlReceiver>();
                if (cc != null)
                {
                    cc.ApplyKnockback(dir, knockDist);
                    cc.ApplyAirborne(1.2f);

                    // Wall-slam detection — check if target lands near a wall
                    Vector3 predictedLand = col.transform.position + dir * knockDist;
                    if (Physics.CheckSphere(predictedLand, 1.0f, wallMask))
                    {
                        float wallDmg = 100f + rank * 35f;
                        hp.ApplyPhysicalDamage(ApplyPen(wallDmg));
                        cc.ApplyStun(0.6f);
                        combatVfx.PlayWallSlam(predictedLand);
                    }
                }
                GainAdrenaline(heroDefinition.spartanKickAdrenaline);
                kicked = true;
            }
            elapsed += Time.deltaTime;
            yield return null;
        }

        agent.Warp(transform.position);
        agent.isStopped = false;
    }

    // ── ULTIMATE — Staff of Hermes: full adrenaline cost + massive AoE ────
    public bool TryCastUltimate()
    {
        if (isDead || ultimateCooldown > 0f) return false;
        if (currentAdrenaline < stats.maxAdrenaline) return false; // must be full

        ultimateCooldown    = heroDefinition.ultimate.CooldownAtRank(ultimateRank);
        currentAdrenaline   = 0f; // drain entire bar
        isUltimateChanneling = true;
        agent.isStopped     = true;
        animator.SetTrigger("CastUltimate");
        StartCoroutine(IshuWrathRoutine(ultimateRank));
        return true;
    }

    private IEnumerator IshuWrathRoutine(int rank)
    {
        // Wind-up: Staff materialises
        combatVfx.PlayStaffSummon(staffMount.position);
        yield return new WaitForSeconds(1.0f);

        float blastRadius  = 14f + rank * 2f;
        float blastDmg     = heroDefinition.ultimate.DamageAtRank(rank, GetEffectiveAttack());
        float fieldBurnDps = 90f + rank * 40f;
        float fieldDuration = 4f + (rank == 3 ? 1f : 0f);

        combatVfx.PlayIshuBlast(transform.position, blastRadius);

        // Burst damage + airborne launch
        foreach (var col in Physics.OverlapSphere(transform.position, blastRadius, enemyMask))
        {
            col.GetComponent<Health>()?.ApplyPhysicalDamage(ApplyPen(blastDmg));
            col.GetComponent<CrowdControlReceiver>()?.ApplyAirborne(1.5f);
        }

        // Sustained ground field (ticks every 0.5s)
        StartCoroutine(IshuGroundFieldRoutine(transform.position, blastRadius, fieldBurnDps, fieldDuration));

        yield return new WaitForSeconds(0.4f);
        isUltimateChanneling = false;

                half3 color = lerp(_EdgeColor.rgb, _CoreColor.rgb, centreW) * _Intensity;
                float alpha = centreW * (0.55 + 0.45 * band) * _Alpha * _CoreColor.a;

                return half4(color, alpha);
            }
            ENDHLSL
        }
    }
}
